"""Integer Newton-Raphson sqrt for u128 inputs.

Clone of deepbook_predict::math::sqrt_u128 + sqrt_initial_guess_u128.
See shared/svi-spec.md §5 "Integer Newton sqrt" for the locked algorithm.

Source: scripts/deepbookv3/packages/predict/sources/helper/math.move:266-292
SHA: 1159d79af33c70e09e406310e1d8f067832ede9d
"""


def isqrt_initial_guess(x: int) -> int:
    """Bit-length-based initial guess: 1 << ((bits + 1) // 2).

    Mirrors helper/math.move:271-292 sqrt_initial_guess_u128 bit-shift sequence.
    """
    if x == 0:
        return 0
    bits = 0
    val = x
    if val >= 1 << 64:
        val >>= 64
        bits += 64
    if val >= 1 << 32:
        val >>= 32
        bits += 32
    if val >= 1 << 16:
        val >>= 16
        bits += 16
    if val >= 1 << 8:
        val >>= 8
        bits += 8
    if val >= 1 << 4:
        val >>= 4
        bits += 4
    if val >= 1 << 2:
        val >>= 2
        bits += 2
    if val >= 1 << 1:
        bits += 1
    return 1 << ((bits + 1) // 2)


def isqrt_u128(x: int) -> int:
    """Integer sqrt: matches on-chain sqrt_u128 bit-for-bit for x in [0, 2^128).

    7 unrolled Newton iterations + overshoot correction. Deterministic, constant-step.
    """
    if x == 0:
        return 0
    if x < 4:
        return 1
    g = isqrt_initial_guess(x)
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    g = (g + x // g) // 2
    if g * g > x:
        g -= 1
    return g
```

"""Raw SVI total variance + binary_price evaluator.

Clones on-chain oracle.move::compute_nd2.
All inputs/outputs at FLOAT_SCALING = 1e9.

Source: scripts/deepbookv3/packages/predict/sources/oracle.move:400-429
SHA: 1159d79af33c70e09e406310e1d8f067832ede9d

from typing import NamedTuple

from .isqrt import isqrt_u128
from .ln import ln_signed
from .phi import normal_cdf

F: int = 1_000_000_000  # FLOAT_SCALING


class SVIParams(NamedTuple):
    a: int          # u64 ≥ 0
    b: int          # u64 ≥ 0
    rho: int        # signed, |rho| < F
    m: int          # signed
    sigma: int      # u64 > 0


def total_variance(svi: SVIParams, k: int) -> int:
    """w(k) = a + b * (rho * (k - m) + sqrt((k - m)^2 + sigma^2))."""
    k_minus_m = k - svi.m  # signed
    k_minus_m_sq = (k_minus_m * k_minus_m) // F  # always non-negative
    sigma_sq = (svi.sigma * svi.sigma) // F
    sq = isqrt_u128((k_minus_m_sq + sigma_sq) * F)  # sqrt at scale F
    rho_km = (svi.rho * k_minus_m) // F  # signed, truncates toward zero
    inner = rho_km + sq  # signed
    if inner < 0:
        raise ValueError("SVI inner term negative — invalid params (assert !is_negative on-chain)")
    w = svi.a + (svi.b * inner) // F
    if w == 0:
        raise ValueError("Total variance is zero (oracle would assert EZeroVariance)")
    return w


def binary_price(svi: SVIParams, forward: int, strike: int) -> int:
    """Theoretical binary call price at FLOAT_SCALING.

    Caller must ensure forward > 0 and strike > 0 (Phase 2 vault.rebalance concern).
    Returns Phi(d2) where d2 = -((k + w/2) / sqrt(w)) and k = ln(strike/forward).
    """
    if forward <= 0:
        raise ValueError("forward must be > 0 (oracle would assert EZeroForward)")
    if strike <= 0:
        raise ValueError("strike must be > 0")
    # k = ln(strike * F / forward) at FLOAT_SCALING; signed
    k = ln_signed((strike * F) // forward)
    w = total_variance(svi, k)
    sqrt_var = isqrt_u128(w * F)  # sqrt of w at FLOAT_SCALING
    half_var = w // 2
    d2_numerator = k + half_var  # signed
    # Signed truncating divide: (d2_numerator * F) // sqrt_var — Python `//` rounds to -inf for
    # negatives; truncate-toward-zero needed per shared/svi-spec.md §2.
    abs_num = abs(d2_numerator)
    d2_mag = (abs_num * F) // sqrt_var
    d2 = -d2_mag if d2_numerator >= 0 else d2_mag  # negate per oracle.move:425 `i64::neg(&d2)`
    # NOTE: oracle.move computes d2 = -((k + w/2)/sqrt_var); so positive d2_numerator gives negative d2.
    return normal_cdf(d2)
```
