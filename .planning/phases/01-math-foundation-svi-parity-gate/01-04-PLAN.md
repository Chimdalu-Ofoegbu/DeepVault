#!/usr/bin/env python3
"""Golden-vector emitter: paper inputs + stress generator + JackJacquier fixture
+ Predict-tests fixture -> shared/golden-vectors.json + contracts/tests/golden_vectors_data.move.

The Python canonical evaluator (deepvault.svi) IS the ground truth. This script:
  1. Builds an inventory of input parameter sets (Tier A / B / C / C2).
  2. Runs deepvault.svi.total_variance and binary_price on each.
  3. Writes the JSON + Move companion files deterministically (sorted keys, hex integers).

Invocations:
    python scripts/golden_emit.py             # regenerate
    python scripts/golden_emit.py --check     # drift check (exit 1 on diff)

CI integration: ci.yml `codegen-drift` job extended to also verify
shared/golden-vectors.json + contracts/tests/golden_vectors_data.move stay in sync.

NOTE: This script must run inside the backtest workspace's uv env so that
`from deepvault.svi import ...` resolves. Invocation pattern (from CI + Makefile):
    cd backtest && uv run --no-project python ../scripts/golden_emit.py
"""

from __future__ import annotations

# Allow running from any CWD by adding backtest/src to sys.path before importing deepvault

# -------------------------------------------------------------------------
# Tier B: Synthetic stress at parametric grid (≥80 vectors, includes arb-violating sub-tier)
# -------------------------------------------------------------------------

def tier_b_vectors() -> list[dict]:
    """Synthetic stress: grid sweep across (a, b, rho, m, sigma) × k × strike."""
    vectors = []

    # Grid axes (per CONTEXT.md re-route + svi-spec.md §3 max safe input domain):
    # k in {-2.0, -1.5, -1.0, -0.5, 0, 0.5, 1.0, 1.5, 2.0} * 1e9   = 9 points
    # rho in {-700M, -300M, 0, +300M, +700M}                         = 5 values
    # b   in {100M, 500M, 1000M}                                     = 3 values
    # sigma in {100M, 500M}                                          = 2 values
    # a fixed at 50M; m fixed at 0
    # Cross-product = 9 * 5 * 3 * 2 = 270 combinations. Sub-sample to ~70 to keep total at ~120.
    counter = 1
    sample_step = 4  # take every 4th combination to land ~70 vectors
    combos = []
    for k_int in [-2_000_000_000, -1_500_000_000, -1_000_000_000, -500_000_000, 0, 500_000_000, 1_000_000_000, 1_500_000_000, 2_000_000_000]:
        for rho in [-700_000_000, -300_000_000, 0, 300_000_000, 700_000_000]:
            for b in [100_000_000, 500_000_000, 1_000_000_000]:
                for sigma in [100_000_000, 500_000_000]:
                    combos.append((50_000_000, b, rho, 0, sigma, k_int))
    # Sample every Nth combination
    sampled = combos[::sample_step]
    for (a, b, rho, m, sigma, k) in sampled:
        # Compute strike from k: k = ln(strike/forward), so strike = forward * exp(k)
        # For golden-vector simplicity, set forward=50*F and pick strike s.t. ln(strike*F/forward) ≈ k
        # at integer-int precision. Use binary_price's ln signal: compute strike = forward * exp(k/F)
        # → since exp is float here for vector generation, do an integer approximation:
        # strike = forward * (F + k) // F   (linear approx of exp; OK because k is small)
        # OR more accurately, use Python `math.exp` ONLY in the emitter (NOT in the evaluator!) to compute strike.
        import math  # ALLOWED in emitter only — not in deepvault.svi
        strike = round(50 * F * math.exp(k / F))
        try:
            w = total_variance(SVIParams(a, b, rho, m, sigma), k)
            bp = binary_price(SVIParams(a, b, rho, m, sigma), forward=50 * F, strike=strike)
            params_valid = True
        except ValueError:
            # Inner negative or zero variance — params reject path
            w = 0
            bp = 0
            params_valid = False
        vectors.append(_make_vector_explicit(
            id_=f"B-{counter:03d}", tier="B", source="synthetic-stress-grid",
            inputs={"a": a, "b": b, "rho": rho, "m": m, "sigma": sigma, "k": k,
                    "T_seconds": 1209600, "forward": 50 * F, "strike": strike},
            expected={"w": w, "binary_price": bp, "params_valid": params_valid,
                      "min_g_k": 0, "calendar_pass": True},  # min_g_k filled by Plan 01-08
        ))
        counter += 1

    # Arb-violating sub-tier (≥10): rho near boundary + extreme params
    arb_violating = [
        # eta * (1 + |rho|) > 2 in SSVI form ⇒ raw-SVI inner term goes negative for some k
        (5_000_000, 1_500_000_000, -990_000_000, 0, 50_000_000, 800_000_000),  # extreme rho
        (5_000_000, 1_500_000_000, -990_000_000, 0, 50_000_000, 1_500_000_000),
        (10_000_000, 2_000_000_000, +990_000_000, 0, 100_000_000, -1_500_000_000),
        (10_000_000, 2_000_000_000, +990_000_000, 0, 100_000_000, -2_000_000_000),
        (0, 3_000_000_000, -999_000_000, 0, 1_000_000, 1_000_000_000),  # a=0 + extreme
        (0, 3_000_000_000, +999_000_000, 0, 1_000_000, -1_000_000_000),
        (50_000_000, 4_000_000_000, -800_000_000, 500_000_000, 50_000_000, -500_000_000),
        (50_000_000, 4_000_000_000, +800_000_000, -500_000_000, 50_000_000, 500_000_000),
        (10_000_000, 5_000_000_000, -950_000_000, 0, 10_000_000, 1_500_000_000),
        (10_000_000, 5_000_000_000, +950_000_000, 0, 10_000_000, -1_500_000_000),
    ]
    for (a, b, rho, m, sigma, k) in arb_violating:
        try:
            w = total_variance(SVIParams(a, b, rho, m, sigma), k)
            params_valid = True   # if we got here, the params actually didn't reject — adjust input
            bp = 0  # not computed for non-rejecting case
        except ValueError:
            w = 0
            bp = 0
            params_valid = False
        vectors.append(_make_vector_explicit(
            id_=f"B-arb-{counter:03d}", tier="B", source="synthetic-arb-violating",
            inputs={"a": a, "b": b, "rho": rho, "m": m, "sigma": sigma, "k": k,
                    "T_seconds": 1209600, "forward": 50 * F, "strike": 50 * F},
            expected={"w": w, "binary_price": bp, "params_valid": params_valid,
                      "min_g_k": -1, "calendar_pass": True},  # min_g_k stubbed; Plan 01-08 fills
        ))
        counter += 1

    return vectors

def _make_vector(*, id_: str, tier: str, source: str, a: int, b: int, rho: int, m: int,
                 sigma: int, k: int, forward: int, strike: int) -> dict:
    """Build a vector dict by running deepvault.svi on the inputs."""
    svi = SVIParams(a=a, b=b, rho=rho, m=m, sigma=sigma)
    try:
        w = total_variance(svi, k)
        bp = binary_price(svi, forward, strike)
        params_valid = True
    except ValueError:
        w = 0
        bp = 0
        params_valid = False
    return _make_vector_explicit(
        id_=id_, tier=tier, source=source,
        inputs={"a": a, "b": b, "rho": rho, "m": m, "sigma": sigma, "k": k,
                "T_seconds": 1209600, "forward": forward, "strike": strike},
        expected={"w": w, "binary_price": bp, "params_valid": params_valid,
                  "min_g_k": 0, "calendar_pass": True},
    )


def _to_signed_hex_pair(value: int) -> dict:
    """Encode signed int as {magnitude_hex, is_negative} pair (matches Move i64::I64).

    Per shared/svi-spec.md §7 sign convention: zero is positive (is_negative=false).
    """
    is_neg = value < 0
    mag = -value if is_neg else value
    return {"mag": hex(mag), "neg": is_neg}


def _make_vector_explicit(*, id_: str, tier: str, source: str,
                          inputs: dict, expected: dict) -> dict:
    """Encode a vector dict to the canonical JSON schema (hex strings + signed pairs)."""
    enc_inputs = {}
    for key in ["a", "b", "sigma", "k", "forward", "strike", "m", "rho"]:
        v = inputs[key]
        if key in ("rho", "m", "k"):  # signed
            enc_inputs[key] = _to_signed_hex_pair(v)
        else:
            enc_inputs[key] = hex(v)
    enc_inputs["T_seconds"] = inputs["T_seconds"]

    enc_expected = {
        "w": hex(expected["w"]),
        "binary_price": hex(expected["binary_price"]),
        "params_valid": expected["params_valid"],
        "min_g_k": _to_signed_hex_pair(expected["min_g_k"]) if isinstance(expected["min_g_k"], int) else expected["min_g_k"],
        "calendar_pass": expected["calendar_pass"],
    }
    return {
        "id": id_,
        "tier": tier,
        "source": source,
        "inputs": enc_inputs,
        "expected": enc_expected,
    }


def emit_json(vectors: list[dict]) -> str:
    return json.dumps(vectors, indent=2, sort_keys=True, ensure_ascii=False) + "\n"


def emit_move_data(vectors: list[dict]) -> str:
    """Emit Move companion data file with vector_count + per-field accessors."""
    parts = []
    parts.append("// =" + "=" * 75)
    parts.append("\n// AUTO-GENERATED - DO NOT EDIT")
    parts.append("\n// Source: shared/golden-vectors.json (regenerated by scripts/golden_emit.py)")
    parts.append("\n// =" + "=" * 75)
    parts.append("\n\nmodule deepvault::golden_vectors_data;\n\n")
    parts.append("// Vector count\n")
    parts.append(f"public fun vector_count(): u64 {{ {len(vectors)} }}\n\n")

    def _decode_signed(enc: dict) -> tuple[int, bool]:
        mag = int(enc["mag"], 16)
        neg = enc["neg"]
        return mag, neg

    # Each vector packed as `vector[a, b, rho_mag, rho_neg_bool_as_u8, m_mag, m_neg, sigma, k_mag, k_neg, forward, strike]`
    parts.append("/// Returns one vector<u64> per golden vector with packed inputs:\n")
    parts.append("/// [a, b, rho_mag, rho_neg(0|1), m_mag, m_neg(0|1), sigma, k_mag, k_neg(0|1), forward, strike]\n")
    parts.append("public fun all_inputs(): vector<vector<u64>> {\n    vector[\n")
    for v in vectors:
        a = int(v["inputs"]["a"], 16)
        b = int(v["inputs"]["b"], 16)
        rho_mag, rho_neg = _decode_signed(v["inputs"]["rho"])
        m_mag, m_neg = _decode_signed(v["inputs"]["m"])
        sigma = int(v["inputs"]["sigma"], 16)
        k_mag, k_neg = _decode_signed(v["inputs"]["k"])
        forward = int(v["inputs"]["forward"], 16)
        strike = int(v["inputs"]["strike"], 16)
        parts.append(
            f"        vector[{a}, {b}, {rho_mag}, {1 if rho_neg else 0}, "
            f"{m_mag}, {1 if m_neg else 0}, {sigma}, "
            f"{k_mag}, {1 if k_neg else 0}, {forward}, {strike}],   // {v['id']} {v['source']}\n"
        )
    parts.append("    ]\n}\n\n")

    parts.append("public fun all_expected_w(): vector<u64> {\n    vector[")
    for v in vectors:
        parts.append(f" {int(v['expected']['w'], 16)},")
    parts.append(" ]\n}\n\n")

    parts.append("public fun all_expected_binary_price(): vector<u64> {\n    vector[")
    for v in vectors:
        parts.append(f" {int(v['expected']['binary_price'], 16)},")
    parts.append(" ]\n}\n\n")

    parts.append("public fun all_params_valid(): vector<bool> {\n    vector[")
    for v in vectors:
        parts.append(f" {'true' if v['expected']['params_valid'] else 'false'},")
    parts.append(" ]\n}\n")

    return "".join(parts)


def write(path: Path, content: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(content, encoding="utf-8", newline="\n")


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--check", action="store_true", help="Drift check only - exit 1 on diff")
    parser.add_argument("--first", type=int, default=None, help="Only first N vectors (debug)")
    args = parser.parse_args()

    vectors = tier_a_vectors() + tier_b_vectors() + tier_c_vectors() + tier_c2_vectors()
    if args.first:
        vectors = vectors[: args.first]

    json_text = emit_json(vectors)
    move_text = emit_move_data(vectors)

    if args.check:
        any_drift = False
        for path, expected in ((JSON_PATH, json_text), (MOVE_DATA_PATH, move_text)):
            actual = path.read_text(encoding="utf-8") if path.exists() else ""
            if actual != expected:
                print(f"DRIFT: {path.relative_to(REPO_ROOT)}", file=sys.stderr)
                any_drift = True
        return 1 if any_drift else 0

    write(JSON_PATH, json_text)
    write(MOVE_DATA_PATH, move_text)
    print(f"golden_emit.py: wrote {len(vectors)} vectors to {JSON_PATH.relative_to(REPO_ROOT)}")
    print(f"golden_emit.py: wrote Move companion to {MOVE_DATA_PATH.relative_to(REPO_ROOT)}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
