module deepvault::helpers::math;

const EZeroDivisor: u64 = 0;

public fun mul_div_round_down(a: u64, b: u64, c: u64): u64 {
    assert!(c > 0, EZeroDivisor);
    (((a as u128) * (b as u128)) / (c as u128)) as u64
}

public fun mul_div_round_up(a: u64, b: u64, c: u64): u64 {
    assert!(c > 0, EZeroDivisor);
    let prod = (a as u128) * (b as u128);
    let div = (c as u128);
    let q = prod / div;
    let r = prod % div;
    let q_round = if (r > 0) { q + 1 } else { q };
    q_round as u64
}
```

// Cloned from helper/math.move:266-292 (sqrt_u128 + initial guess) and :120-125 (sqrt wrapper).
// SHA: 1159d79af33c70e09e406310e1d8f067832ede9d
// Algorithm locked in shared/svi-spec.md §5: bit-length seed + 7 unrolled Newton iterations + overshoot.
module deepvault::helpers::isqrt;

const EInvalidPrecision: u64 = 0;
const FLOAT_SCALING: u64 = 1_000_000_000;

public fun sqrt_u128(x: u128): u128 {
    if (x == 0) return 0;
    if (x < 4) return 1;
    let mut g = sqrt_initial_guess_u128(x);
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    g = (g + x / g) / 2;
    if (g * g > x) { g = g - 1; };
    g
}

public fun sqrt_initial_guess_u128(x: u128): u128 {
    let mut bits: u8 = 0;
    let mut val = x;
    if (val >= 1u128 << 64) { val = val >> 64; bits = bits + 64; };
    if (val >= 1u128 << 32) { val = val >> 32; bits = bits + 32; };
    if (val >= 1u128 << 16) { val = val >> 16; bits = bits + 16; };
    if (val >= 1u128 << 8) { val = val >> 8; bits = bits + 8; };
    if (val >= 1u128 << 4) { val = val >> 4; bits = bits + 4; };
    if (val >= 1u128 << 2) { val = val >> 2; bits = bits + 2; };
    if (val >= 1u128 << 1) { bits = bits + 1; };
    1u128 << (((bits + 1) / 2) as u8)
}

/// FLOAT_SCALING-aware sqrt wrapper: x and precision both at scale FLOAT_SCALING.
public fun sqrt(x: u64, precision: u64): u64 {
    assert!(precision > 0 && precision <= FLOAT_SCALING, EInvalidPrecision);
    let multiplier = (FLOAT_SCALING / precision) as u128;
    let scaled = (x as u128) * multiplier * (FLOAT_SCALING as u128);
    (sqrt_u128(scaled) / multiplier) as u64
}
```

    // a signed result already at FLOAT_SCALING. Match the on-chain behavior exactly.
    let k = ln::ln(math::mul_div_round_down(strike, FLOAT_SCALING, forward));
    binary_price_from_k(a, b, rho, m, sigma, &k)

/// Innermost evaluator: takes pre-computed signed k. Used internally by both entries above
/// AND by total_variance_from_params test entry.
fun binary_price_from_k(a: u64, b: u64, rho: I64, m: I64, sigma: u64, k: &I64): u64 {
    // Range-check k (Phase 2 vault enforces this; svi_view does it as defense-in-depth).
    assert!(i64::magnitude(k) <= strategy_constants::svi_k_max_log_strike(), EKOutOfRange);

    let total_var = total_variance_from_params(a, b, rho, m, sigma, k);
    assert!(total_var > 0, EZeroVariance);

    let sqrt_var = isqrt::sqrt(total_var, FLOAT_SCALING);
    let sqrt_var_i64 = i64::from_u64(sqrt_var);
    let half_var_i64 = i64::from_u64(total_var / 2);

    let d2_intermediate = i64::div_scaled(&d2_numerator, &sqrt_var_i64);  // (num * F) / sqrt_var, signed
    let d2 = i64::neg(&d2_intermediate);  // negate per oracle.move:425

}
