// dashboard/src/lib/math.ts
// Canonical mul-div helper. ALL svi/phi/isqrt/ln arithmetic uses this; never inline `a * b / c`.
// See shared/svi-spec.md §2 "Op-order canonical form".
// Mirrors deepvault::helpers::math::mul_div_round_down (Move).

export const FLOAT_SCALING: bigint = 1_000_000_000n;

export function mulDivRoundDown(a: bigint, b: bigint, c: bigint): bigint {
  if (c === 0n) {
    throw new Error('mulDivRoundDown: divisor is zero');
  }
  return (a * b) / c;
}

export function mulDivRoundUp(a: bigint, b: bigint, c: bigint): bigint {
  if (c === 0n) {
    throw new Error('mulDivRoundUp: divisor is zero');
  }
  const prod = a * b;
  const q = prod / c;
  const r = prod % c;
  return r > 0n ? q + 1n : q;
}

// dashboard/src/lib/isqrt.ts
// Integer Newton-Raphson sqrt — bigint clone of helper/math.move:266-292.
// Algorithm locked in shared/svi-spec.md §5: bit-length seed + 7 unrolled Newton + overshoot.
// Source: scripts/deepbookv3/packages/predict/sources/helper/math.move:266-292
// SHA: 1159d79af33c70e09e406310e1d8f067832ede9d

export function isqrtInitialGuess(x: bigint): bigint {
  if (x === 0n) return 0n;
  let bits = 0n;
  let val = x;
  if (val >= 1n << 64n) { val >>= 64n; bits += 64n; }
  if (val >= 1n << 32n) { val >>= 32n; bits += 32n; }
  if (val >= 1n << 16n) { val >>= 16n; bits += 16n; }
  if (val >= 1n << 8n)  { val >>= 8n;  bits += 8n; }
  if (val >= 1n << 4n)  { val >>= 4n;  bits += 4n; }
  if (val >= 1n << 2n)  { val >>= 2n;  bits += 2n; }
  if (val >= 1n << 1n)  {              bits += 1n; }
  return 1n << ((bits + 1n) / 2n);
}

export function isqrtU128(x: bigint): bigint {
  if (x === 0n) return 0n;
  if (x < 4n) return 1n;
  let g = isqrtInitialGuess(x);
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  g = (g + x / g) / 2n;
  if (g * g > x) g -= 1n;
  return g;
}
```

// dashboard/src/lib/svi.ts
// Raw 5-parameter SVI evaluator — bigint clone of Python deepvault.svi (Plan 01-03) and
// Move deepvault::svi_view (Plan 01-05). All clones derived from oracle.move:400-429.
//
// Forbidden: Number(), parseFloat, Math.X.
//
// Source: scripts/deepbookv3/packages/predict/sources/oracle.move:400-429
// SHA: 1159d79af33c70e09e406310e1d8f067832ede9d

import { FLOAT_SCALING, mulDivRoundDown } from './math';
import { isqrtU128 } from './isqrt';
import { lnSigned } from './ln';
import { normalCdf } from './phi';
import { STRATEGY_CONSTANTS } from './strategy_constants';

const F = FLOAT_SCALING;  // 1_000_000_000n

export type SVIParams = {
  a: bigint;
  b: bigint;
  rho: bigint;    // signed bigint
  m: bigint;      // signed bigint
  sigma: bigint;
};

/// Validate SVIParams against the bounds in shared/strategy.toml [svi].
function _validateParams(svi: SVIParams): void {
  if (svi.a > STRATEGY_CONSTANTS.SVI_A_MAX) throw new Error('a out of range');
  if (svi.b > STRATEGY_CONSTANTS.SVI_B_MAX) throw new Error('b out of range');
  if (svi.sigma < STRATEGY_CONSTANTS.SVI_SIGMA_MIN || svi.sigma > STRATEGY_CONSTANTS.SVI_SIGMA_MAX) {
    throw new Error('sigma out of range');
  }
  const rhoMag = svi.rho < 0n ? -svi.rho : svi.rho;
  if (rhoMag >= F) throw new Error('|rho| must be < F');
  const mMag = svi.m < 0n ? -svi.m : svi.m;
  if (mMag > STRATEGY_CONSTANTS.SVI_M_ABS_MAX) throw new Error('|m| out of range');
}

/// Total variance w(k) = a + b * (rho * (k - m) + sqrt((k - m)^2 + sigma^2)).
export function totalVariance(svi: SVIParams, k: bigint): bigint {
  const kMinusM = k - svi.m;
  const kMinusMSq = (kMinusM * kMinusM) / F;
  const sigmaSq = mulDivRoundDown(svi.sigma, svi.sigma, F);
  const sq = isqrtU128((kMinusMSq + sigmaSq) * F);
  // rho * (k-m) at scale F: signed mul-div with truncate-toward-zero.
  const absKMinusM = kMinusM < 0n ? -kMinusM : kMinusM;
  const absRho = svi.rho < 0n ? -svi.rho : svi.rho;
  const rhoKmMag = (absRho * absKMinusM) / F;
  const rhoKmSign = (svi.rho < 0n) !== (kMinusM < 0n);  // XOR signs
  const rhoKm = rhoKmSign ? -rhoKmMag : rhoKmMag;
  const inner = rhoKm + sq;
  if (inner < 0n) {
    throw new Error('SVI inner term negative — invalid params');
  }
  const w = svi.a + (svi.b * inner) / F;
  if (w === 0n) {
    throw new Error('Total variance is zero');
  }
  return w;
}

/// Theoretical binary call price at FLOAT_SCALING.
export function binaryPrice(svi: SVIParams, forward: bigint, strike: bigint): bigint {
  if (forward <= 0n) throw new Error('forward must be > 0');
  if (strike <= 0n) throw new Error('strike must be > 0');
  _validateParams(svi);

  // k = lnSigned(strike * F / forward); range-check
  const ratio = mulDivRoundDown(strike, F, forward);
  const k = lnSigned(ratio);
  const kMag = k < 0n ? -k : k;
  if (kMag > STRATEGY_CONSTANTS.SVI_K_MAX_LOG_STRIKE) {
    throw new Error('k out of safe input domain (svi-spec.md §3)');
  }

  const w = totalVariance(svi, k);
  const sqrtVar = isqrtU128(w * F);
  const halfVar = w / 2n;
  const d2Numerator = k + halfVar;  // signed
  // Signed truncating divide: (d2Numerator * F) / sqrtVar
  const absNum = d2Numerator < 0n ? -d2Numerator : d2Numerator;
  const d2Mag = (absNum * F) / sqrtVar;
  const d2Sign = d2Numerator < 0n;
  // d2 = -((k + halfVar) / sqrtVar)  →  if d2Numerator >= 0, d2 is negative; if < 0, d2 is positive
  const d2 = d2Sign ? d2Mag : -d2Mag;
  return normalCdf(d2);
}

/// Convenience entry returning both outputs.
export function evaluateSVI(
  svi: SVIParams,
  forward: bigint,
  strike: bigint,
): { w: bigint; binaryPrice: bigint } {
  const ratio = mulDivRoundDown(strike, F, forward);
  const k = lnSigned(ratio);
  const w = totalVariance(svi, k);
  const bp = binaryPrice(svi, forward, strike);
  return { w, binaryPrice: bp };
}
```
