```python
"""Arb-free checker: closed-form butterfly bound + 200-point g(k) grid scan + calendar test.

Off-chain only (Move evaluator hard-rejects on closed-form alone per CONTEXT.md D-05).
Returns full g(k) array for dashboard visualization (MATH-04 lever).

Source: Gatheral & Jacquier (2014) §3.2 + RESEARCH.md Pattern 4 lines 432-441.

Type discipline (Pitfall A): numpy allowed for vector grid sampling, but the OUTPUT g_k_array
is converted to Python int before returning. Parity_runner asserts isinstance(elem, int).
"""
from __future__ import annotations

from typing import NamedTuple

import numpy as np  # ALLOWED HERE (visualization-bound, not parity-bound)

from .strategy_constants import (
    SVI_GRID_POINTS_FOR_ARB_CHECK,
    SVI_STRIKE_RANGE_SIGMA,
)
from .svi import SVIParams, total_variance

F: int = 1_000_000_000  # FLOAT_SCALING


class ArbResult(NamedTuple):
    params_valid: bool         # True iff min(g(k)) >= 0 across the grid
    min_g_k: int               # at FLOAT_SCALING; if negative → arb violation
    calendar_pass: bool        # single-tenor stub: True with documented reason
    g_k_array: list[int]        # length >= SVI_GRID_POINTS_FOR_ARB_CHECK; int at FLOAT_SCALING


def check_arb(svi: SVIParams) -> ArbResult:
    """Compute the g(k) curve over ±strike_range_sigma * sqrt(theta_T_estimate) and check >= 0.

    Args:
        svi: raw SVI parameters at FLOAT_SCALING.

    Returns:
        ArbResult with full g(k) array (Phase 4 dashboard renders this as a curve).
    """
    n = SVI_GRID_POINTS_FOR_ARB_CHECK
    # Estimate theta_T (ATM total variance) from svi.a + svi.b * sigma (raw SVI ATM proxy):
    # At k = m, w(m) = a + b * (rho * 0 + sqrt(0 + sigma^2)) = a + b * sigma / 1.0 (at scale F)
    # = svi.a + (svi.b * svi.sigma) // F
    theta_t_atm = svi.a + (svi.b * svi.sigma) // F
    if theta_t_atm <= 0:
        # Degenerate — no defined ATM variance; treat as invalid
        return ArbResult(params_valid=False, min_g_k=-F, calendar_pass=True, g_k_array=[0] * n)

    # k_range = ±strike_range_sigma * sqrt(theta_t_atm) at FLOAT_SCALING
    # sqrt(theta_t_atm) (in 1e9) is approximately sqrt(theta_t_atm * F) // F (handled by isqrt)
    from .isqrt import isqrt_u128  # local import to avoid evaluator namespace pollution
    sqrt_theta_t = isqrt_u128(theta_t_atm * F)  # at scale F
    k_max = SVI_STRIKE_RANGE_SIGMA * sqrt_theta_t

    # Build the grid (numpy ALLOWED here for vectorized sampling).
    k_grid_np = np.linspace(-k_max, k_max, n).astype(np.int64)
    # Evaluate w(k), w'(k), w''(k) and then g(k) at each k. Use closed-form derivatives:
    # t = k - m; r = sqrt((k - m)^2 + sigma^2)
    # w(k) = a + b * (rho * t + r); w'(k) = b * (rho + t / r); w''(k) = b * sigma^2 / r^3
    # g(k) = (1 - k * w'(k) / (2 * w(k)))^2 - (w'(k))^2 / 4 * (1 / w(k) + 1/4) + w''(k) / 2
    #
    # The grid evaluation uses Python int (NOT numpy float) inside the loop to maintain bit-equal
    # output with the totalVariance call. We use numpy ONLY for the linspace generation.
    g_k = []
    for k_val in k_grid_np.tolist():  # convert to Python int via .tolist()
        try:
            w = total_variance(svi, int(k_val))
        except ValueError:
            # Inner negative — arb violation triggers via the evaluator's own reject path
            g_k.append(-F)
            continue

        # Compute derivatives manually (mirroring the formulas above; integer arithmetic)
        t = int(k_val) - svi.m
        t_sq = (t * t) // F
        sigma_sq = (svi.sigma * svi.sigma) // F
        r_sq_at_F = (t_sq + sigma_sq) * F  # input to isqrt
        r = isqrt_u128(r_sq_at_F)  # u64 at scale F; r >= 0 always

        # w'(k) = b * (rho + t / r) at scale F
        # rho is signed; t/r is signed
        if r == 0:
            g_k.append(-F)
            continue
        # signed divide: (t * F) // r with sign tracked
        t_div_r = (abs(t) * F) // r * (1 if t >= 0 else -1)
        w_prime = (svi.b * (svi.rho + t_div_r)) // F  # signed

        # w''(k) = b * sigma_sq / r^3 (all at scale F, careful with the cube)
        # r^3 at F: r * r // F * r // F = (r * r // F) * r // F   (clamp to avoid overflow)
        r_sq = (r * r) // F
        r_cubed = (r_sq * r) // F
        if r_cubed == 0:
            g_k.append(-F)
            continue
        w_double_prime = (svi.b * sigma_sq * F) // r_cubed  # all unsigned

        # g(k) at scale F:
        # term1 = (1 - k * w' / (2 * w))^2
        # term2 = -(w')^2 / 4 * (1/w + 1/4)
        # term3 = w''/2
        if w == 0:
            g_k.append(-F)
            continue
        # k * w' / (2 * w) at scale F — signed
        kwp_2w_mag = (abs(int(k_val)) * abs(w_prime) * F) // (2 * w * F)  # = abs(k * w_prime) // (2 * w)
        kwp_2w_sign = (int(k_val) < 0) ^ (w_prime < 0)
        kwp_2w = -kwp_2w_mag if kwp_2w_sign else kwp_2w_mag

        one_minus = F - kwp_2w  # 1 at scale F minus signed
        term1_unsigned = (one_minus * one_minus) // F  # squared, always >= 0

        # (w')^2 / 4
        wp_sq = (w_prime * w_prime) // F  # unsigned (squared)
        # (1 / w + 1/4) at scale F: F * F / w + F / 4
        one_over_w = (F * F) // w
        one_quarter = F // 4
        sum_term = one_over_w + one_quarter
        term2 = (wp_sq * sum_term) // (F * 4)  # divide by 4 separately to avoid overflow

        term3 = w_double_prime // 2

        g = term1_unsigned - term2 + term3
        g_k.append(int(g))

    min_g = min(g_k)
    return ArbResult(
        params_valid=min_g >= 0,
        min_g_k=int(min_g),
        calendar_pass=True,  # single-tenor; svi-spec.md §"calendar pass stub"
        g_k_array=g_k,
    )
```

"""Tests for arb_checker: closed-form + g(k) grid + Tier C JackJacquier cross-check."""
import json
from pathlib import Path

import pytest

from deepvault.arb_checker import check_arb, ArbResult
from deepvault.svi import SVIParams, total_variance
from deepvault.strategy_constants import SVI_GRID_POINTS_FOR_ARB_CHECK

F: int = 1_000_000_000


def test_returns_arb_result_with_correct_shape():
    svi = SVIParams(a=50_000_000, b=250_000_000, rho=0, m=0, sigma=400_000_000)
    result = check_arb(svi)
    assert isinstance(result, ArbResult)
    assert isinstance(result.params_valid, bool)
    assert isinstance(result.min_g_k, int)
    assert isinstance(result.calendar_pass, bool)
    assert isinstance(result.g_k_array, list)
    assert len(result.g_k_array) >= SVI_GRID_POINTS_FOR_ARB_CHECK
    for g in result.g_k_array:
        assert isinstance(g, int)
        assert not isinstance(g, bool)


def test_valid_slice_passes():
    """Sane SVI params should produce min_g_k >= 0."""
    svi = SVIParams(a=50_000_000, b=250_000_000, rho=-100_000_000, m=0, sigma=400_000_000)
    result = check_arb(svi)
    assert result.params_valid is True
    assert result.min_g_k >= 0


def test_extreme_rho_violates():
    """Extreme rho close to ±1 with low sigma should produce min_g_k < 0 somewhere."""
    svi = SVIParams(a=10_000_000, b=2_000_000_000, rho=-995_000_000, m=0, sigma=10_000_000)
    result = check_arb(svi)
    # Either params_valid=False (g(k) dipped below 0 on grid) OR a ValueError caught upstream
    assert result.params_valid is False or any(g < 0 for g in result.g_k_array)


def test_calendar_pass_is_stub_true():
    """Single-tenor: calendar_pass returns True with documented reason in svi-spec.md."""
    svi = SVIParams(a=50_000_000, b=250_000_000, rho=0, m=0, sigma=400_000_000)
    result = check_arb(svi)
    assert result.calendar_pass is True


def test_g_k_array_visualization_data():
    """Plan 01-08's MATH-04 lever: the array IS the visualization data Phase 4 dashboard renders."""
    svi = SVIParams(a=50_000_000, b=250_000_000, rho=-100_000_000, m=0, sigma=400_000_000)
    result = check_arb(svi)
    # Array has minimum length, integer-typed elements, and finite values
    assert len(result.g_k_array) >= 200
    for g in result.g_k_array:
        assert isinstance(g, int)
        assert -1_000_000_000_000 < g < 1_000_000_000_000  # reasonable range

```typescript
// dashboard/src/lib/arb_checker.ts
// Off-chain arbitrage-free checker — bigint mirror of backtest/src/deepvault/arb_checker.py.
//
// Returns full g(k) array for Phase 4 dashboard visualization (MATH-04 lever).
//
// Note: TypeScript bigint is arbitrary-precision (like Python int) so no numpy equivalent
// is needed. The grid is built via a simple loop; no external dep beyond bigint primitives.

import { totalVariance, type SVIParams } from './svi';
import { isqrtU128 } from './isqrt';
import { FLOAT_SCALING, mulDivRoundDown } from './math';
import { STRATEGY_CONSTANTS } from './strategy_constants';

const F = FLOAT_SCALING;  // 1_000_000_000n

export type ArbResult = {
  paramsValid: boolean;
  minGk: bigint;
  calendarPass: boolean;
  gK: bigint[];
};

export function checkArb(svi: SVIParams): ArbResult {
  const n = STRATEGY_CONSTANTS.SVI_GRID_POINTS_FOR_ARB_CHECK;  // = 200
  const strikeRangeSigma = BigInt(STRATEGY_CONSTANTS.SVI_STRIKE_RANGE_SIGMA);  // = 4n

  // Estimate theta_t_atm = a + (b * sigma) / F
  const thetaTAtm = svi.a + (svi.b * svi.sigma) / F;
  if (thetaTAtm <= 0n) {
    return {
      paramsValid: false,
      minGk: -F,
      calendarPass: true,
      gK: new Array(n).fill(0n),
    };
  }
  const sqrtTheta = isqrtU128(thetaTAtm * F);
  const kMax = strikeRangeSigma * sqrtTheta;

  // Build the linear grid: k_i = -kMax + i * (2 * kMax) / (n - 1) for i in [0, n)
  const gK: bigint[] = [];
  const step = (2n * kMax) / BigInt(n - 1);
  for (let i = 0; i < n; i++) {
    const k = -kMax + BigInt(i) * step;
    let w: bigint;
    try {
      w = totalVariance(svi, k);
    } catch {
      gK.push(-F);
      continue;
    }
    // t = k - m; r = sqrt((k-m)^2 + sigma^2) at scale F
    const t = k - svi.m;
    const tSq = (t * t) / F;
    const sigmaSq = mulDivRoundDown(svi.sigma, svi.sigma, F);
    const rSqAtF = (tSq + sigmaSq) * F;
    const r = isqrtU128(rSqAtF);
    if (r === 0n) {
      gK.push(-F);
      continue;
    }
    // signed divide: (t * F) / r
    const absT = t < 0n ? -t : t;
    const tDivRMag = (absT * F) / r;
    const tDivR = t < 0n ? -tDivRMag : tDivRMag;
    const wPrime = (svi.b * (svi.rho + tDivR)) / F;
    // r^3 at F
    const rSq = (r * r) / F;
    const rCubed = (rSq * r) / F;
    if (rCubed === 0n) {
      gK.push(-F);
      continue;
    }
    const wDoublePrime = (svi.b * sigmaSq * F) / rCubed;
    if (w === 0n) {
      gK.push(-F);
      continue;
    }
    // k * w' / (2 * w) signed
    const absK = k < 0n ? -k : k;
    const absWp = wPrime < 0n ? -wPrime : wPrime;
    const kwp2wMag = (absK * absWp * F) / (2n * w * F);
    const kwp2wSign = (k < 0n) !== (wPrime < 0n);
    const kwp2w = kwp2wSign ? -kwp2wMag : kwp2wMag;
    const oneMinus = F - kwp2w;
    const term1Unsigned = (oneMinus * oneMinus) / F;
    // (w')^2 / 4 * (1/w + 1/4)
    const wpSq = (wPrime * wPrime) / F;
    const oneOverW = (F * F) / w;
    const oneQuarter = F / 4n;
    const sumTerm = oneOverW + oneQuarter;
    const term2 = (wpSq * sumTerm) / (F * 4n);
    const term3 = wDoublePrime / 2n;
    const g = term1Unsigned - term2 + term3;
    gK.push(g);
  }

  let minG = gK[0];
  for (const g of gK) {
    if (g < minG) minG = g;
  }
  return {
    paramsValid: minG >= 0n,
    minGk: minG,
    calendarPass: true,
    gK,
  };
}
```
