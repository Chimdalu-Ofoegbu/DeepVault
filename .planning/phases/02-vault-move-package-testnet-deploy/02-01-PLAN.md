    ```bash
    #!/usr/bin/env bash
    # scripts/verify-deepbookv3-pin.sh
    # Asserts that contracts/Move.toml's DeepBookV3 `rev = "..."` matches the
    # upstream SHA the vendored subtree at scripts/deepbookv3/ was last pulled from.
    # Mitigates RESEARCH.md Pitfall 6 (SHA pin drift between Move.toml and subtree).
    #
    # Exit codes:
    #   0  Pin matches subtree last-pull SHA
    #   1  Pin differs (drift detected)
    #   2  Cannot determine subtree pull SHA (unexpected repo state)

    set -euo pipefail

    REPO_ROOT="$(git rev-parse --show-toplevel)"
    cd "$REPO_ROOT"

    # 1. Extract the rev from Move.toml.
    MOVE_TOML_REV="$(grep -E 'DeepBookV3.*rev\s*=' contracts/Move.toml | sed -E 's/.*rev\s*=\s*"([0-9a-f]+)".*/\1/')"
    if [[ -z "${MOVE_TOML_REV}" ]] || [[ ! "${MOVE_TOML_REV}" =~ ^[0-9a-f]{40}$ ]]; then
      echo "::error::Could not parse a 40-char SHA rev from contracts/Move.toml DeepBookV3 line."
      exit 2
    fi

    # 2. Extract the upstream SHA the subtree was last squashed from.
    # `git log` on the subtree path with --follow shows the most recent merge/squash;
    # the commit body of a `git subtree --squash` commit contains "git-subtree-split: <upstream-sha>".
    SUBTREE_SPLIT="$(git log --pretty=format:'%b' -- scripts/deepbookv3 | grep -E '^git-subtree-split:' | head -n 1 | awk '{print $2}')"
    if [[ -z "${SUBTREE_SPLIT}" ]] || [[ ! "${SUBTREE_SPLIT}" =~ ^[0-9a-f]{40}$ ]]; then
      echo "::error::Could not extract git-subtree-split SHA from scripts/deepbookv3 history."
      echo "          Was the subtree pulled with --squash? Run scripts/predict-diff.sh to investigate."
      exit 2
    fi

    # 3. Compare.
    if [[ "${MOVE_TOML_REV}" != "${SUBTREE_SPLIT}" ]]; then
      echo "::error::DeepBookV3 SHA pin drift detected."
      echo "          Move.toml rev:           ${MOVE_TOML_REV}"
      echo "          Subtree last-pull SHA:   ${SUBTREE_SPLIT}"
      echo "          Fix: bump one or both. See CONTRIBUTING.md weekly Monday Predict sweep."
      exit 1
    fi

    echo "OK: DeepBookV3 pin aligned at ${MOVE_TOML_REV}"
    exit 0
    ```
