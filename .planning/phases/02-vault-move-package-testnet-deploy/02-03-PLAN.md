## W1: Lock the FINAL Vault<Quote> struct schema upfront

    admin: address,
    /// D-10: pause halts supply only; redeems and rolls always flow.
    paused: bool,
    /// Liquid quote balance (DUSDC on testnet, USDsui on mainnet).

    escrow_balance: Balance<SHARE>,
    /// Per-user redeem-request slots (D-02 — per-user independent slot, NO global queue).

## W2: Lock RequestSlot to the Balance<SHARE> form

The W2 fix selects `Balance<SHARE>` over the u64-with-escrow_balance alternative. The locked struct is:

```move
public struct RequestSlot has store {
    /// Escrowed SHARE balance — the user's shares are physically held here, not just counted.
    /// Cancel returns this directly via `balance.into_coin(ctx)` for a clean Coin<SHARE>.
    shares_escrowed: Balance<SHARE>,
    /// Wall-clock at which redeem_request was called; cooldown gate D-01 = 1h.
    request_timestamp_ms: u64,
    /// Tracks how much of the original escrow has been paid out across multiple
    /// redeem_fulfill calls (D-03 liquidity-short fulfill — partial payouts allowed).
    claimed_so_far: u64,
}
```

`Balance<SHARE>` does NOT have `drop`, so RequestSlot does NOT have `drop`. Constructors and destructors:
- `new_request_slot(shares: Balance<SHARE>, timestamp_ms: u64): RequestSlot`
- `request_destroy(slot: RequestSlot): Balance<SHARE>` — caller must consume the returned Balance (e.g., via `coin::from_balance`).
- `request_split_shares(slot: &mut RequestSlot, amount: u64): Balance<SHARE>` — for partial fulfill.
- `request_shares_value(slot: &RequestSlot): u64` — read accessor.

The `escrow_balance: Balance<SHARE>` field on Vault is REDUNDANT under the W2 lock (the Balance lives directly in each RequestSlot). KEEP it anyway — it provides a fast-path accessor for total escrowed shares without iterating request_slots, which the dashboard subscribes to. Initialize to `balance::zero()`. Each `redeem_request` MUST `escrow_balance.join(shares.split(amount))` AND store the split balance in the slot — i.e., the `escrow_balance` is the SUM-OF-RECORDS while each slot owns its individual portion. Plan 02-05's redeem.move maintains this invariant.

If maintaining the dual-bookkeeping invariant proves cumbersome, Plan 02-05 may revert `escrow_balance` to a u64 counter (sum of slot.shares_escrowed.value() across all slots) and keep the per-slot Balance as the authoritative store. Document the choice in the 02-05 SUMMARY. Plan 02-03 ships both forms; Plan 02-05 collapses to one.

    // Copyright (c) DeepVault.
    // SPDX-License-Identifier: MIT
    //
    // VAULT_SHARE coin type. TreasuryCap quarantine via PendingTreasury bridge:
    // the cap is wrapped into PendingTreasury at init() and only consumed once
    // by deepvault::vault::create_vault. Closes VAULT-02.

    module deepvault::share;

    use sui::coin::TreasuryCap;
    use sui::coin_registry;

    /// One-time witness; Move guarantees init() is called exactly once.
    public struct SHARE has drop {}

    /// Bridge struct moving TreasuryCap from init() to vault::create_vault.
    /// Has key+store so init() can transfer it; vault::create_vault unpacks
    /// it and moves the inner cap into the shared Vault<Quote> struct.
    public struct PendingTreasury has key, store {
        id: UID,
        cap: TreasuryCap<SHARE>,
    }

    fun init(witness: SHARE, ctx: &mut TxContext) {
        let (mut initializer, treasury_cap) = coin_registry::new_currency_with_otw(
            witness,
            9,                                  // share_decimals (Phase 0 D-13)
            b"dvUSDC".to_string(),              // testnet symbol; mainnet: dvUSDsui

            b"Vault share for DeepVault PLP+Hedge".to_string(),
            b"".to_string(),                    // icon URL deferred to Phase 5
            ctx,
        );
        let metadata_cap = initializer.finalize(ctx);
        transfer::public_transfer(metadata_cap, ctx.sender());
        transfer::public_transfer(
            PendingTreasury { id: object::new(ctx), cap: treasury_cap },
            ctx.sender(),
        );
    }

    /// Unpacks the PendingTreasury for vault::create_vault. `public(package)`
    /// keeps it inside the deepvault package — no external module can extract.
    public(package) fun consume_pending(pending: PendingTreasury): TreasuryCap<SHARE> {
        let PendingTreasury { id, cap } = pending;
        id.delete();
        cap

    #[test_only]

    public fun create_vault<Quote>(
        pending: PendingTreasury,
        seed: Coin<Quote>,
        clock: &Clock,
        ctx: &mut TxContext,
    ) {
        let cap = share::consume_pending(pending);

        assert!(
            seed.value() == strategy_constants::seed_quote_micro_units(),
            ESeedAmountMismatch,
        );

        let predict_manager_id = deepbook_predict::predict_manager::new(ctx);
        let admin_addr = ctx.sender();

        let mut vault = Vault<Quote> {
            id: object::new(ctx),
            treasury_cap: cap,
            admin: admin_addr,
            paused: false,
            balance: balance::zero(),
            escrow_balance: balance::zero(),
            request_slots: table::new(ctx),
            rate_limiters: table::new(ctx),
            hedges: table::new(ctx),
            hedge_keys: vector[],
            predict_manager_id,
            total_shares_supply: 0,
            total_assets: 0,
            // Initialize tunable_* to strategy_constants defaults.
            tunable_token_bucket_capacity_quote_micro_units: strategy_constants::token_bucket_capacity(),
            tunable_token_bucket_refill_rate_quote_micro_units_per_ms: strategy_constants::token_bucket_refill_rate_per_ms(),
            tunable_hedge_policy_allocation_bps: strategy_constants::allocation_bps(),
            tunable_hedge_policy_strike_otm_bps: strategy_constants::strike_otm_bps(),
            tunable_hedge_policy_tenor_seconds: strategy_constants::tenor_seconds(),
            tunable_oracle_max_staleness_seconds: strategy_constants::max_staleness_seconds(),
        };

        // Seed: mint virtual_shares and burn to @0xdead; lock seed quote into balance.
        let seed_shares_amt = strategy_constants::virtual_shares();
        let seed_shares = coin::mint(&mut vault.treasury_cap, seed_shares_amt, ctx);
        transfer::public_transfer(seed_shares, DEAD_ADDRESS);

        let seed_amt = seed.value();
        vault.balance.join(seed.into_balance());
        vault.total_assets = seed_amt;
        vault.total_shares_supply = seed_shares_amt;

        let vault_id = object::id(&vault);

        let admin_cap = AdminCap { id: object::new(ctx) };
        transfer::transfer(admin_cap, admin_addr);
        transfer::share_object(vault);

        event::emit(VaultCreated {
            vault_id,
            admin: admin_addr,
            seed_quote: seed_amt,
            seed_shares_burned: seed_shares_amt,
            predict_manager_id,
        });
    }
