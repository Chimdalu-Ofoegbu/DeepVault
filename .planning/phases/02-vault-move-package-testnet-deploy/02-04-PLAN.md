    ```move
    // Copyright (c) DeepVault.
    // SPDX-License-Identifier: MIT
    //
    // Atomic supply path: deposit Coin<Quote>, mint Coin<SHARE>, purchase
    // binary hedge via predict_adapter::mint — all in one Move transaction.
    // Closes VAULT-03 + the supply portion of VAULT-05.

    module deepvault::supply;

    use deepvault::{
        vault::{Self, Vault},
        share::SHARE,
        rebalance,
        strategy_constants,
    };
    use deepvault::helpers::math;
    use sui::{coin::{Self, Coin}, clock::Clock, event};

    // Error codes 200-299 reserved for supply.move.
    const ESupplyPaused: u64 = 200;
    const EZeroSharesMinted: u64 = 201;
    const EShareOverflow: u64 = 202;
    const EZeroAmount: u64 = 203;

    public struct Supplied has copy, drop, store {
        vault_id: ID,
        depositor: address,
        deposit_quote: u64,
        shares_minted: u64,
        hedge_alloc_quote: u64,
    }

    /// Atomic supply + hedge entry. `public fun` (NOT `entry`) per RESEARCH.md
    /// Pattern 3 — keeps PTB chaining open for Phase 4's two-protocol PTB.
    public fun supply<Quote>(
        vault: &mut Vault<Quote>,
        predict: &mut deepbook_predict::predict::Predict,
        predict_manager: &mut deepbook_predict::predict_manager::PredictManager,
        oracle: &deepbook_predict::oracle::OracleSVI,
        deposit: Coin<Quote>,
        clock: &Clock,
        ctx: &mut TxContext,
    ) {
        assert!(!vault::is_paused(vault), ESupplyPaused);
        let amount = deposit.value();
        assert!(amount > 0, EZeroAmount);

        let shares_to_mint = compute_shares_to_mint(vault, amount);
        assert!(shares_to_mint > 0, EZeroSharesMinted);

        // Split: 90% to vault.balance, 10% to hedge purchase (D-01 / Phase 0).
        let hedge_alloc_bps = strategy_constants::allocation_bps();
        let hedge_amount = math::mul_div_round_down(amount, hedge_alloc_bps, 10_000);
        let mut deposit_balance = deposit.into_balance();
        let hedge_balance = deposit_balance.split(hedge_amount);

        // Atomic hedge purchase. Aborts whole tx on Predict's 30s staleness gate
        // (oracle_config::assert_live_oracle) or our max_price_premium_bps abstain.
        let hedge_alloc_value = hedge_balance.value();
        rebalance::buy_hedge_for_deposit<Quote>(
            vault, predict, predict_manager, oracle,
            coin::from_balance(hedge_balance, ctx),
            clock, ctx,
        );

        // Update accounting
        let post_balance_amount = deposit_balance.value();
        vault::balance_mut(vault).join(deposit_balance);
        vault::add_total_assets(vault, amount); // total deposit (incl. hedge alloc — vault accounts for both legs)
        vault::add_total_shares(vault, shares_to_mint);

        // Mint share coin to depositor
        let share_coin = coin::mint(vault::treasury_cap_mut(vault), shares_to_mint, ctx);
        transfer::public_transfer(share_coin, ctx.sender());

        event::emit(Supplied {
            vault_id: object::id(vault),
            depositor: ctx.sender(),
            deposit_quote: amount,
            shares_minted: shares_to_mint,
            hedge_alloc_quote: hedge_alloc_value,
        });
    }

    /// Virtual-shares math (OpenZeppelin ERC-4626 v5).
    /// Truncate-toward-zero division rounds DOWN, in vault's favor (PITFALLS Pitfall 12).
    public(package) fun compute_shares_to_mint<Quote>(
        vault: &Vault<Quote>,
        deposit: u64,
    ): u64 {
        let virtual_shares = strategy_constants::virtual_shares();
        let numerator = (deposit as u128)
            * ((vault::total_shares(vault) as u128) + (virtual_shares as u128));
        let denominator = (vault::total_assets(vault) as u128) + 1u128;
        let shares = numerator / denominator;
        assert!(shares <= (std::u64::max_value!() as u128), EShareOverflow);
        shares as u64
    }
    ```

    ```move
        oracle: &OracleSVI,

    /// else push new entry.
    public(package) fun insert_or_consolidate_hedge<Quote>(
        vault: &mut Vault<Quote>,

        oracle_id: ID,
        strike: u64,
        expiry_ms: u64,
        quantity: u64,
        cost_basis: u64,
    ) {

            vault::hedges_mut(vault).add(key, h);
            vector::push_back(vault::hedge_keys_mut(vault), key);
        };
    }
    ```

    ```move
    /// NAV per share at 1e9 fixed-point. Sum of liquid quote + future:hedge book
    /// value (v1: hedge cost basis carried at acquisition, no SVI re-mark on this
    /// path — full SVI-marked NAV is a future v2 enhancement consumed by the
    /// dashboard).
    public fun nav_per_share<Quote>(vault: &Vault<Quote>): u64 {
        let total_shares = vault::total_shares(vault);
        assert!(total_shares > 0, EZeroShares);
        // total_assets includes hedge cost basis (per supply.move: add_total_assets uses full deposit amount).
        math::mul_div_round_down(
            vault::total_assets(vault),
            strategy_constants::nav_scale(),
            total_shares,
        )
    }

    /// D-14 pessimistic: assumes ALL open hedges expire worthless.
    /// Output: u64 NAV per share at 1e9 fixed-point (D-15).
    /// Consumed by future Margin liquidation path.
    public fun worst_case_nav_per_share<Quote>(vault: &Vault<Quote>): u64 {
        let total_shares = vault::total_shares(vault);
        assert!(total_shares > 0, EZeroShares);
        // D-14: only liquid balance counts (all hedges go to zero).
        // Note: balance_value is the QUOTE balance held inside vault (excludes
        // amount sent to Predict for hedge purchases — that's already gone).
        math::mul_div_round_down(
            vault::balance_value(vault),
            strategy_constants::nav_scale(),
            total_shares,
        )
    }

    /// Convenience: returns the haircut as bps from current NAV.
    /// haircut_bps = 10_000 * (nav - worst_case_nav) / nav
    public fun worst_case_haircut_bps<Quote>(vault: &Vault<Quote>): u64 {
        let nav = nav_per_share(vault);
        if (nav == 0) { return 0 };
        let worst = worst_case_nav_per_share(vault);
        // worst <= nav by construction; safe subtraction.
        math::mul_div_round_down(nav - worst, 10_000, nav)
    }
    ```
