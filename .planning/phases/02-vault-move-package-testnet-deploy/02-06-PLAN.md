    ```move
    // ============================================================
    // AdminCap-gated entry functions (D-11). AdminCap is `key`-only,
    // non-transferable in v1 (D-12). No admin_transfer_cap exists.
    // ============================================================

    // Error codes 600-699 reserved for admin functions.
    const EUnknownTuneKey: u64 = 600;

    public struct Paused has copy, drop, store { vault_id: ID, paused: bool }
    public struct AdminOverride has copy, drop, store {
        vault_id: ID, parameter: vector<u8>, old_value: u64, new_value: u64,
    }
    public struct AdminTune has copy, drop, store {
        vault_id: ID, key: std::string::String, old_value: u64, new_value: u64,
    }
    public struct AdminUnwind has copy, drop, store {
        vault_id: ID, market_key: deepbook_predict::market_key::MarketKey,
    }

    /// D-11.1: toggle the pause flag. Pause halts supply only — redeem and
    /// roll keep working (D-10).
    public fun admin_pause<Quote>(
        vault: &mut Vault<Quote>,
        _cap: &AdminCap,
        paused: bool,
    ) {
        vault.paused = paused;
        event::emit(Paused { vault_id: object::id(vault), paused });
    }

    /// D-11.2: runtime override of vault.max_staleness_seconds.
    ///
    /// IMPORTANT: This is a DISPLAY-ONLY parameter. Predict's on-chain
    /// `oracle_config::assert_live_oracle` (predict-internal) enforces a
    /// HARD 30-second staleness ceiling at predict::mint call time. AdminCap
    /// CANNOT relax that gate. This setting only changes what the dashboard
    /// shows as "expected stale" — actual mints fail when the oracle is older
    /// than 30 seconds, regardless of this value. (RESEARCH.md note 1.)
    public fun admin_oracle_staleness_override<Quote>(
        vault: &mut Vault<Quote>,
        _cap: &AdminCap,
        max_seconds: u64,
    ) {
        let old = vault.max_staleness_seconds;
        vault.max_staleness_seconds = max_seconds;
        event::emit(AdminOverride {
            vault_id: object::id(vault),
            parameter: b"max_staleness_seconds",
            old_value: old,
            new_value: max_seconds,
        });
    }

    /// D-11.3: runtime tune one of five strategy parameters.
    public fun admin_tune_strategy<Quote>(
        vault: &mut Vault<Quote>,
        _cap: &AdminCap,
        key: std::string::String,
        value: u64,
    ) {
        let key_bytes = key.as_bytes();
        let old: u64;
        if (key_bytes == &b"[token_bucket].capacity") {
            old = vault.tunable_token_bucket_capacity;
            vault.tunable_token_bucket_capacity = value;
        } else if (key_bytes == &b"[token_bucket].refill_rate") {
            old = vault.tunable_token_bucket_refill_rate_per_ms;
            vault.tunable_token_bucket_refill_rate_per_ms = value;
        } else if (key_bytes == &b"[hedge_policy].ratio_bps") {
            old = vault.tunable_hedge_alloc_bps;
            vault.tunable_hedge_alloc_bps = value;
        } else if (key_bytes == &b"[hedge_policy].strike_otm_bps") {
            old = vault.tunable_strike_otm_bps;
            vault.tunable_strike_otm_bps = value;
        } else if (key_bytes == &b"[hedge_policy].tenor_seconds") {
            old = vault.tunable_tenor_seconds;
            vault.tunable_tenor_seconds = value;
        } else {
            abort EUnknownTuneKey
        };
        event::emit(AdminTune {
            vault_id: object::id(vault),
            key,
            old_value: old,
            new_value: value,
        });
    }

    /// D-11.4: close one specific open hedge regardless of expiry.
    /// Narrow but load-bearing for "we can pull the plug if oracle is wrong".
    public fun admin_emergency_unwind<Quote>(
        vault: &mut Vault<Quote>,
        _cap: &AdminCap,
        predict: &mut deepbook_predict::predict::Predict,
        predict_manager: &mut deepbook_predict::predict_manager::PredictManager,
        oracle: &deepbook_predict::oracle::OracleSVI,
        market_key: deepbook_predict::market_key::MarketKey,
        clock: &sui::clock::Clock,
        ctx: &mut TxContext,
    ) {
        // Remove hedge from registry.
        let hedge = vault.hedges.remove(market_key);
        let (found, idx) = vector::index_of(&vault.hedge_keys, &market_key);
        if (found) { vector::remove(&mut vault.hedge_keys, idx); };

        // Close via adapter.
        deepvault::predict_adapter::redeem<Quote>(
            predict, predict_manager, oracle, market_key,
            hedge_quantity(&hedge), clock, ctx,
        );

        event::emit(AdminUnwind { vault_id: object::id(vault), market_key });
    }
    ```
