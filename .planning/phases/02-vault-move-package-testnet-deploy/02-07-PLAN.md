    // three specs use it; if not, refactor to field-read capture.

    // Closes VAULT-10 (inflation safety property).

    module deepvault::specs::inflation_safe;

    use deepvault::{vault::Vault, supply, strategy_constants};

    /// Minimum deposit below which we accept that shares_to_mint may round
    /// to zero (denial of microscopic deposits is not an inflation attack).
    /// Set to 1_000 micro-units (= 0.001 DUSDC at 6dp).
    const MIN_DEPOSIT_THRESHOLD: u64 = 1_000;

    #[spec(prove)]
    fun shares_to_mint_positive_for_meaningful_deposit<Quote>(
        vault: &Vault<Quote>,
        deposit: u64,
    ): u64 {
        // Preconditions: vault is seeded (post-create_vault state).
        requires(vault::total_assets(vault) >= strategy_constants::seed_quote_micro_units());
        requires(vault::total_shares(vault) >= strategy_constants::virtual_shares());
        requires(deposit >= MIN_DEPOSIT_THRESHOLD);

        // Computed value (the function under test).
        let shares = supply::compute_shares_to_mint(vault, deposit);

        // Property: shares > 0 (inflation defense holds for meaningful deposits).
        ensures(shares > 0);

        shares
    }

    **File 2: `contracts/specs/nav_monotone.move`**:

    ```move
    // Copyright (c) DeepVault.
    // SPDX-License-Identifier: MIT
    //
    // Sui Prover spec — NAV per share never decreases on supply.
    // Property: Calling supply with a positive deposit cannot decrease
    // nav_per_share (modulo hedge_cost_tolerance: hedge cost is 10% of
    // deposit and is expected to come back as fair-value hedge book asset,
    // so net NAV change ~ 0 within a per-block tolerance).
    //
    // Closes VAULT-10 (NAV monotonicity property).

    module deepvault::specs::nav_monotone;

    use deepvault::{vault::Vault, ltv, strategy_constants};

    /// Tolerance: hedge cost rounding + minor SVI fair-value drift.
    /// At 1e9 fixed-point, allow a 1bp slip per supply call.
    fun hedge_cost_tolerance_x9(): u64 {
        // 1bp on a 1e9-scaled NAV-per-share = 100_000.
        100_000
    }

    /// SPIKE OUTCOME: If clone!(vault) compiles and runs, use it. If not,
    /// capture pre-state via field reads — store old_total_assets +
    /// old_total_shares as requires() arithmetic.
    ///
    /// Choice of B (field reads) is more portable; choice of A (clone!)
    /// is preferred if Sui Prover supports it.

    #[spec(prove)]
    fun nav_does_not_decrease_on_state_change<Quote>(
        old_total_assets: u64,
        old_total_shares: u64,
        new_total_assets: u64,
        new_total_shares: u64,
    ) {
        // Preconditions: positive shares pre and post.
        requires(old_total_shares > 0);
        requires(new_total_shares > 0);
        // Supply increases both total_assets and total_shares by positive amounts.
        requires(new_total_assets >= old_total_assets);
        requires(new_total_shares >= old_total_shares);

        // Compute NAV per share at 1e9 fixed-point pre and post.
        let old_nav = ((old_total_assets as u128) * (strategy_constants::nav_scale() as u128))
            / (old_total_shares as u128);
        let new_nav = ((new_total_assets as u128) * (strategy_constants::nav_scale() as u128))
            / (new_total_shares as u128);

        // Postcondition: NAV does not decrease beyond tolerance.
        ensures(new_nav + (hedge_cost_tolerance_x9() as u128) >= old_nav);
    }
    ```

    This formulation models the NAV transition as a pure-arithmetic property — SAFER for Sui Prover than the runtime-state version. The "real" supply call is exercised by tests in supply_test.move; the spec proves the math is sound for any monotone state transition.
