    let admin = @0xa1;
    let user = @0xb2;
    let mut scenario = ts::begin(admin);

    // 1. Setup: share::init then vault::create_vault<TEST_QUOTE> with 10 DUSDC seed.
    share::init_for_testing(scenario.ctx());

    let seed_amt = strategy_constants::seed_quote_micro_units();
    let seed = coin::mint_for_testing<TEST_QUOTE>(seed_amt, scenario.ctx());
    let mut clk = clock::create_for_testing(scenario.ctx());
    vault::create_vault<TEST_QUOTE>(pending, seed, &clk, scenario.ctx());

    // 2. Mint 100_000 SHARE to user via test-only mint helper (added if missing
    //    to vault.move under #[test_only] visibility; the helper directly bumps
    //    treasury_cap and total_shares_supply, then transfers the Coin<SHARE>
    //    to `user`. If the helper does not exist, add it as part of this task —
    //    `vault::test_mint_shares_to<TEST_QUOTE>(vault: &mut Vault<TEST_QUOTE>, recipient: address, amount: u64, ctx: &mut TxContext)`.)
    scenario.next_tx(admin);
    let mut vault_obj = scenario.take_shared<vault::Vault<TEST_QUOTE>>();
    let user_share_amount = 100_000u64;
    vault::test_mint_shares_to<TEST_QUOTE>(&mut vault_obj, user, user_share_amount, scenario.ctx());
    // Bump total_assets so NAV math is well-defined: deposit 1_000_000 micro-DUSDC into balance.
    let user_quote_deposit = 1_000_000u64;
    let extra_balance = coin::mint_for_testing<TEST_QUOTE>(user_quote_deposit, scenario.ctx());
    vault::balance_mut(&mut vault_obj).join(extra_balance.into_balance());
    vault::add_total_assets(&mut vault_obj, user_quote_deposit);

    // Compute expected_payout BEFORE redeem to capture the pre-state.
    // NAV per share at 1e9 fixed-point:
    //     nav_per_share = (total_assets * 1e9) / total_shares
    //                   = (10_000_000 + 1_000_000) * 1e9 / (1_000_000 + 100_000)
    //                   = 11_000_000_000_000_000_000 / 1_100_000
    //                   = 10_000_000_000  (= 10 quote per share at 1e9 fp)
    // pro-rata payout for 100_000 user shares:
    //     payout = 100_000 * nav_per_share / 1e9 = 1_000_000 micro-DUSDC = 1 DUSDC.
    // Bound by liquid balance (11_000_000) and bucket capacity (default 100 DUSDC = 100_000_000) — payout fits.
    let expected_payout = 1_000_000u64;

    // 3. Hand the user their 100_000 SHARE coin and switch scenario to user.
    ts::return_shared(vault_obj);
    scenario.next_tx(user);
    let user_shares = scenario.take_from_sender<sui::coin::Coin<share::SHARE>>();
    assert_eq!(sui::coin::value(&user_shares), user_share_amount);

    // 4. Call redeem::redeem_request to escrow the shares + record the timestamp.
    let mut vault_obj = scenario.take_shared<vault::Vault<TEST_QUOTE>>();
    redeem::redeem_request<TEST_QUOTE>(&mut vault_obj, user_shares, &clk, scenario.ctx());

    // 5. Warp clock past the 1h cooldown.
    clk.increment_for_testing(redeem::cooldown_ms() + 1);
    // (If `redeem::cooldown_ms()` is not exported, hardcode 3_600_001u64 here and
    //  document the lift in the SUMMARY — but the named accessor is preferred.)

    // 6. Call redeem::redeem_fulfill — pays out min(pro_rata, bucket_avail, liquid).
    redeem::redeem_fulfill<TEST_QUOTE>(&mut vault_obj, &clk, scenario.ctx());

    // 7. Switch back to user inventory and capture the payout Coin<TEST_QUOTE>.
    ts::return_shared(vault_obj);
    scenario.next_tx(user);
    let payout_coin = scenario.take_from_sender<sui::coin::Coin<TEST_QUOTE>>();
    let actual_payout = sui::coin::value(&payout_coin);

    // 8. Assertions (W5 lock — both bounds enforced):
    //    (a) Round-down-in-vault-favor: actual_payout <= expected_payout.
    //    (b) Liquidity bound: actual_payout <= seed + extra_balance pre-fulfill.
    assert!(actual_payout <= expected_payout, /* EPayoutExceedsExpected */ 9001);
    assert!(actual_payout <= (seed_amt + user_quote_deposit), /* EPayoutExceedsLiquid */ 9002);

    // if strategy.toml's tunable_token_bucket_capacity_quote_micro_units < 1_000_000.

    // Cleanup
    sui::test_utils::destroy(payout_coin);
    clk.destroy_for_testing();
    scenario.end();

            }

        FAILED=1
        continue
      fi

      if awk -v c="${COV}" -v t="${THRESHOLD}" 'BEGIN { exit (c+0 >= t+0 ? 0 : 1) }'; then
        echo "OK: ${MOD}.move coverage ${COV}% >= ${THRESHOLD}%"
