### W3 — Port the per-user time-decay RateLimiter into vault_state.py (Task 1)

The Task 1 `redeem_fulfill` body (action body in this plan) currently consumes from the static `TOKEN_BUCKET_CAPACITY` constant via `payable = min(pro_rata_value, self.balance, TOKEN_BUCKET_CAPACITY)`. This is NOT bit-equal to Move's `redeem::redeem_fulfill` (contracts/sources/redeem.move:104-197), which:

1. Lazy-initializes a per-user `RateLimiter` in `get_or_init_user_bucket()` (redeem.move:236-261) using `vault::effective_token_bucket_capacity` and `vault::effective_token_bucket_refill_rate_per_ms`. The bucket starts FULL via `rate_limiter::record_deposit(capacity)`.
2. Computes `bucket_avail = rate_limiter::available_withdrawal(bucket, clock)` (rate_limiter.move:69-80), which time-decays the bucket using `refill_amount = elapsed_ms * refill_rate_per_ms` capped at `capacity`.
3. Consumes via `rate_limiter::consume(bucket, payout, clock)` (rate_limiter.move:97-106), which calls `refill()` first then decrements `available -= amount`.

Without porting this per-user bucket, the Plan 03-06 trace-replay parity test will FAIL the moment the captured testnet trace exercises two redeems by the same user — the second redeem on Move will reflect time-decay refill, but the Python `TOKEN_BUCKET_CAPACITY` constant is stateless and will diverge.

**Amendment scope changes:**

1. **Add `PyRateLimiter` dataclass to `vault_state.py`** (placed AFTER `RequestSlot`, BEFORE `VaultStateSnapshot`). Bit-equal to `deepvault::rate_limiter::RateLimiter` (rate_limiter.move:37-48):

   ```python
   @dataclass
   class PyRateLimiter:
       """Mirrors deepvault::rate_limiter::RateLimiter (contracts/sources/helpers/rate_limiter.move:37-48).

       Bit-equal time-decay token bucket: refills at `refill_rate_per_ms` tokens
       per ms of `elapsed_ms`, capped at `capacity`.

       Source: contracts/sources/helpers/rate_limiter.move
       """
       available: int = 0
       last_refill_ms: int = 0       # mirrors Move's `last_updated_ms`
       tokens_remaining: int = 0     # alias of `available`; kept as separate field so
                                     # the W3 grep gate (`grep -E 'last_refill_ms|tokens_remaining'`)
                                     # has a stable target. Kept in sync with `available`.
       capacity: int = 0
       refill_rate_per_ms: int = 0
       enabled: bool = False

       def refill(self, now_ms: int) -> None:
           """Mirrors rate_limiter::refill (rate_limiter.move:154-164).
           Refills the bucket based on elapsed time, capped at capacity."""
           if now_ms <= self.last_refill_ms:
               return  # mirrors elapsed_ms() guard (rate_limiter.move:167-174)
           elapsed_ms = now_ms - self.last_refill_ms
           if elapsed_ms > 0:
               refill_amount = elapsed_ms * self.refill_rate_per_ms
               new_available = min(self.available + refill_amount, self.capacity)
               self.available = new_available
               self.tokens_remaining = new_available
               self.last_refill_ms = now_ms

       def available_withdrawal(self, now_ms: int) -> int:
           """Mirrors rate_limiter::available_withdrawal (rate_limiter.move:69-80).
           Read-only — does NOT mutate."""
           if not self.enabled:
               # Move returns u64::max_value!() when disabled (rate_limiter.move:70).
               # Use a large sentinel so min(...) in redeem_fulfill behaves correctly.
               return (1 << 64) - 1
           if now_ms <= self.last_refill_ms:
               return self.available
           elapsed_ms = now_ms - self.last_refill_ms
           refill_amount = elapsed_ms * self.refill_rate_per_ms
           return min(self.available + refill_amount, self.capacity)

       def consume(self, amount: int, now_ms: int) -> None:
           """Mirrors rate_limiter::consume (rate_limiter.move:97-106)."""
           if not self.enabled or amount == 0:
               return
           self.refill(now_ms)
           assert amount <= self.capacity, "EExceedsCapacity (0)"
           assert amount <= self.available, "EInsufficientWithdrawalBudget (1)"
           self.available -= amount
           self.tokens_remaining = self.available
   ```

       slot = self.request_slots[user]

       bucket_avail = bucket.available_withdrawal(ts_ms)

       payable = min(pro_rata_value, self.balance, bucket_avail)
       assert payable > 0, "EInsufficientLiquidity (303)"

       # Burn the corresponding share amount; consume from bucket (mutates state).
       if pro_rata_value > 0:

           slot.shares_escrowed -= shares_burned

       # is left untouched so the cooldown remains satisfied.

Mirrors on-chain Move semantics bit-for-bit:
  - vault::create_vault seed transaction (10 DUSDC + 1M virtual shares burned)
  - supply::supply (supply.move:61-117)

  - ltv::nav_per_share (ltv.move:41-58)
  - ltv::worst_case_nav_per_share (ltv.move:60-68)

Forbidden imports: math, numpy, scipy. Pure Python int.

from typing import Dict, NamedTuple, List

from .strategy_constants import (
    VIRTUAL_SHARES,
    SEED_QUOTE_MICRO_UNITS,
    NAV_SCALE,

REDEEM_COOLDOWN_MS: int = 3_600_000  # 1 hour per Phase 2 D-01


@dataclass

    """Mirrors deepvault::rebalance::HedgePosition (vault.move:90-120)."""
    oracle_id: str
    strike: int
    expiry_ms: int
    notional_quote: int
    cost_basis_quote: int
    quantity: int

    """Mirrors deepvault::redeem::RequestSlot per Phase 2 D-02."""
    shares_escrowed: int
    request_timestamp_ms: int
    claimed_so_far: int = 0


class VaultStateSnapshot(NamedTuple):
    """Frozen snapshot used by replay parity (Plan 03-06)."""
    balance: int
    total_assets: int
    total_shares: int


@dataclass
class VaultState:
    """Python mirror of the on-chain Vault<Quote> state.

    Bit-equal to Move semantics for supply, redeem, nav, worst_case_nav.
    Consumed by trace-replay parity (Plan 03-06).
    """
    balance: int = 0
    total_assets: int = 0
    total_shares: int = 0
    paused: bool = False
    hedges: Dict[str, HedgePosition] = field(default_factory=dict)
    request_slots: Dict[str, RequestSlot] = field(default_factory=dict)
    _mock_predict_manager_balance: int = 0  # backtest-only proxy

    def new_seeded(cls) -> 'VaultState':
        """Mirrors vault::create_vault — 10 DUSDC seed + 1M shares burned."""
        v = cls()
        v.balance = SEED_QUOTE_MICRO_UNITS
        v.total_assets = SEED_QUOTE_MICRO_UNITS
        v.total_shares = VIRTUAL_SHARES
        return v

    def snapshot(self) -> VaultStateSnapshot:
        return VaultStateSnapshot(self.balance, self.total_assets, self.total_shares)

    def compute_shares_to_mint(self, deposit: int) -> int:
        """Bit-equal to supply::compute_shares_to_mint (supply.move:143-156)."""
        numerator = deposit * (self.total_shares + VIRTUAL_SHARES)
        denominator = self.total_assets + 1
        return numerator // denominator

    def supply(self, deposit_quote: int) -> int:
        """Mirrors supply::supply (supply.move:61-117). Returns shares minted."""
        assert not self.paused, "ESupplyPaused (200)"
        assert deposit_quote > 0, "EZeroAmount (203)"
        shares = self.compute_shares_to_mint(deposit_quote)
        assert shares > 0, "EZeroSharesMinted (201)"
        hedge_alloc = deposit_quote * ALLOCATION_BPS // 10_000
        self._mock_predict_manager_balance += hedge_alloc
        self.balance += deposit_quote - hedge_alloc
        self.total_assets += deposit_quote
        self.total_shares += shares
        return shares

    def nav_per_share(self) -> int:
        """Bit-equal to ltv::nav_per_share (ltv.move:41-58)."""
        if self.total_shares == 0:
            raise ValueError("EZeroShares (500)")
        return (self.total_assets * NAV_SCALE) // self.total_shares

    def worst_case_nav(self) -> int:
        """Bit-equal to ltv::worst_case_nav_per_share (ltv.move:60-68).
        Uses balance (NOT total_assets) — assumes all hedges expire worthless."""
        if self.total_shares == 0:
            raise ValueError("EZeroShares (500)")
        return (self.balance * NAV_SCALE) // self.total_shares

        """Mirrors redeem::redeem_request (redeem.move per Phase 2 D-02)."""
        assert shares > 0, "EZeroAmount (203)"
        assert user not in self.request_slots, "ERequestSlotExists"
        self.request_slots[user] = RequestSlot(
            shares_escrowed=shares,
            request_timestamp_ms=ts_ms,
            claimed_so_far=0,

    def redeem_fulfill(self, user: str, ts_ms: int) -> int:
        """Mirrors redeem::redeem_fulfill (redeem.move per Phase 2 D-01 + D-03).
        Returns quote micro-units paid out (may be partial per D-03)."""
        assert user in self.request_slots, "ERequestSlotMissing"
        slot = self.request_slots[user]
        elapsed = ts_ms - slot.request_timestamp_ms
        assert elapsed >= REDEEM_COOLDOWN_MS, "ECooldownNotElapsed (D-01)"
        # Pro-rata payout using nav_per_share
        pro_rata_value = (slot.shares_escrowed * self.nav_per_share()) // NAV_SCALE
        # Per D-03: pay min(pro_rata, liquid_balance, bucket_tokens_remaining)
        payable = min(pro_rata_value, self.balance, TOKEN_BUCKET_CAPACITY)
        # Burn the corresponding share amount; remainder stays escrowed with timestamp UNCHANGED
        if payable > 0 and pro_rata_value > 0:
            shares_burned = (payable * NAV_SCALE) // self.nav_per_share()
            slot.shares_escrowed -= shares_burned
            slot.claimed_so_far += payable
            self.balance -= payable
            self.total_assets -= payable
            self.total_shares -= shares_burned
        if slot.shares_escrowed == 0:
            del self.request_slots[user]
        return payable

        """Consumes one action; asserts pre-state matches; applies; asserts post.

        Loop invariant per RESEARCH.md Pitfall 2: the FIRST action's pre must
        match self.new_seeded() exactly; SUBSEQUENT actions' pre must be the
        post of the prior action as computed by THIS VaultState instance. We
        never read pre from the trace to overwrite self.

        Tolerance: 1-wei per CONTEXT.md D-15 / Phase 1 parity discipline.
        """
        pre = {k: int(v) for k, v in action['pre'].items()}
        assert self.balance == pre['balance'], (
            f"pre balance drift: {self.balance} != {pre['balance']}"
        )
        assert self.total_assets == pre['total_assets']
        assert self.total_shares == pre['total_shares']

        kind = action['kind']
        if kind == 'supply':
            self.supply(int(action['args']['deposit_quote']))
        elif kind == 'redeem_request':
            self.redeem_request(
                action['args']['user'],
                int(action['args']['shares']),
                int(action['ts_ms']),
            )
        elif kind == 'redeem_fulfill':
            self.redeem_fulfill(action['args']['user'], int(action['ts_ms']))
        elif kind == 'hedge_mint':
            self._apply_hedge_mint(action)
        elif kind == 'roll':
            self._apply_roll(action)
        else:
            raise ValueError(f"Unknown action kind: {kind!r}")

        post = action['post']
        assert abs(self.balance - int(post['balance'])) <= 1, (
            f"post balance mismatch: {self.balance} vs {post['balance']}"
        )
        assert abs(self.total_assets - int(post['total_assets'])) <= 1
        assert abs(self.total_shares - int(post['total_shares'])) <= 1

    def _apply_hedge_mint(self, action: dict) -> None:
        """Mirrors rebalance::buy_hedge_for_deposit (rebalance.move:219+).
        For now stub: only updates the mock predict manager balance.
        Detailed semantics (HedgePosition registry mutation) land in Plan 03-06
        as the trace replayer fills in the events."""

        new_cost = int(action['args'].get('new_cost_basis_quote', 0))
        self.balance += payoff - new_cost
        self.total_assets += payoff - new_cost

"""Lookahead-bias audit harness — BACK-06, D-06, D-07, D-08.

Type discipline: lookahead_audit is one of the few numpy-allowed modules in
the Python evaluator codebase. numpy is used for np.random.permutation (D-06
shuffled-label test) and np.random.choice (D-07 3-row hand recompute seed).
OUTPUT goes back to pure-Python int / list[int] / float before crossing the
module boundary.

This module does NOT mirror Move semantics (no parity discipline applies).

Per CONTEXT.md:
  D-06: shuffled-label test must produce |alpha| <= 0.5% APY to pass.
  D-07: hand recompute on 3 random trade rows (np.random.choice + seed=42).
  D-08: @strategy_fn decorator (Plan 03-02) enforces decision-bar / observation-bar split.
"""
from typing import Callable, Iterable, FrozenSet

import numpy as np  # ALLOWED HERE (audit-bound, not parity-bound)
import pandas as pd


def pick_hand_recompute_rows(returns: pd.Series, n: int = 3, seed: int = 42) -> list[int]:
    """Per D-07: pick `n` random row indices for hand recompute.

    Deterministic via the given seed (default 42 per CONTEXT.md).
    """
    rng = np.random.default_rng(seed)
    indices = rng.choice(len(returns), size=n, replace=False)
    return [int(i) for i in indices]


def compound_to_apy(total_return: float, bars: int, bars_per_year: int = 8760) -> float:
    """Annualize a cumulative return.

    Per RESEARCH.md A9: 8,760 bars/year is the canonical hourly convention for
    BTC (trades 24/7 — not 252 trading days).
    """
    if bars == 0:
        return 0.0
    return (1.0 + total_return) ** (bars_per_year / bars) - 1.0


def shuffled_label_alpha_apy(
    simulation_fn: Callable,
    returns: pd.Series,
    seed: int = 42,
) -> float:
    """Per D-06: shuffle labels, re-run strategy, compute APY of result.

    A strategy with no lookahead bias should produce ~zero APY on shuffled labels
    because timing — the only edge — is destroyed by shuffling.

    Required gate: |alpha_apy| <= 0.005 (0.5% APY). Anything higher means a leak.
    """
    rng = np.random.default_rng(seed)
    shuffled = pd.Series(rng.permutation(returns.values), index=returns.index)
    result = simulation_fn(shuffled)
    apy = compound_to_apy(result.total_return, result.bars)
    return float(apy)


def inspect_strategy_fn_decls(fn) -> tuple[FrozenSet[str], FrozenSet[str]]:
    """Read the (reads, writes) declarations from a @strategy_fn-decorated function.

    Audit harness uses this to enumerate the read/write declarations across all
    strategy functions before running the full backtest (catches forgotten
    declarations at audit time, not at first DataFrame access).
    """
    if not hasattr(fn, '_reads') or not hasattr(fn, '_writes'):
        raise AttributeError(f"{fn.__name__} is not @strategy_fn-decorated")
    return fn._reads, fn._writes
