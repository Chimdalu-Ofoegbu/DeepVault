const SUPPLIER: address = @0xB2;

// === Helpers ===
fun new_seeded_vault(scenario: &mut ts::Scenario): (Vault<TEST_QUOTE>, AdminCap, sui::clock::Clock) {
    // Mirror integration_test.move:81-94 — share OTW + pending treasury consume + seed coin
    share::init_for_testing(scenario.ctx());
    scenario.next_tx(ADMIN);
    let pending = scenario.take_from_sender<share::PendingTreasury>();

    let seed_amt = strategy_constants::seed_quote_micro_units();

    let (vault, admin_cap) = vault::new_vault_for_testing<TEST_QUOTE>(cap, seed, scenario.ctx());
    let clock = sui::clock::create_for_testing(scenario.ctx());
    (vault, admin_cap, clock)
}

fun cleanup(vault: Vault<TEST_QUOTE>, admin_cap: AdminCap, clock: sui::clock::Clock) {
    vault::destroy_for_testing(vault);
    vault::destroy_admin_cap_for_testing(admin_cap);

}

fun seeded_vault_does_not_expose_treasury_cap() {

def _grep(pattern: str, root: Path, glob: str = '*.move') -> list[str]:
    """Run grep across `root` for `pattern`; return matching lines (excluding test files)."""
    result = subprocess.run(
        ['grep', '-rnE', pattern, str(root), '--include', glob],
        capture_output=True,
        text=True,
    )
    # grep exits 1 when no matches found; that's success for us.
    lines = result.stdout.splitlines()
    return lines
