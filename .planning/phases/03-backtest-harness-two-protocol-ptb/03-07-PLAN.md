```move
fun new_seeded_vault(scenario: &mut ts::Scenario): (Vault<TEST_QUOTE>, AdminCap, Clock) {
```

```python
"""Python ↔ Move parity at the liquidation path (PTB-05 cross-language).

Mirrors contracts/tests/liquidation_test.move's worst_case_nav_at_minus_30_shock
test. Asserts that VaultState.worst_case_nav() produces the same integer value
at the same shocked state, within 1 wei tolerance.

Per CONTEXT.md D-20: supply 1000 DUSDC → buy hedge at SVI fair value →
simulate Predict resolution where binary expires worthless AND vault collateral
drops 30%. Assert worst_case_nav_per_share matches Python within 1 wei.

Per planner critical constraint #7: 1-wei tolerance.
"""
import pytest

from deepvault.vault_state import VaultState
from deepvault.strategy_constants import (
    VIRTUAL_SHARES,
    SEED_QUOTE_MICRO_UNITS,
    NAV_SCALE,
)


def _simulate_supply_1000_dusdc(vault: VaultState) -> None:
    """Simulate the post-supply state of a 1000 DUSDC deposit.

    Mirrors liquidation_test.move's `inflate_liquid_for_testing` + manual
    state mutation. We bypass the supply() formula and set the fields
    directly so that the Python test matches the Move test 1:1.
    """
    # Move test: vault::inflate_liquid_for_testing(&mut vault, 900_000_000) + mint 100M shares
    # Mirror: vault.balance += 900_000_000; total_shares += 100_000_000
    # (total_assets is NOT incremented in the Move helper — it's an arithmetic shortcut)
    vault.balance += 900_000_000
    vault.total_shares += 100_000_000


def test_worst_case_nav_at_minus_30_shock_drops_to_70pct():
    """Mirror liquidation_test.move::worst_case_nav_at_minus_30_shock_drops_to_70pct.
    Python worst_case_nav at -30% shock equals 70% of pre-shock value within 1 wei."""
    v = VaultState.new_seeded()
    _simulate_supply_1000_dusdc(v)

    wcn_pre = v.worst_case_nav()
    # Apply -30% shock: drain 30% of balance
    drain = v.balance * 3_000 // 10_000
    v.balance -= drain

    wcn_post = v.worst_case_nav()
    expected_post = wcn_pre * 7_000 // 10_000
    assert abs(wcn_post - expected_post) <= 1, (
        f"-30% shock arithmetic off: wcn_pre={wcn_pre}, wcn_post={wcn_post}, "
        f"expected={expected_post}, diff={abs(wcn_post - expected_post)}"
    )


def test_python_worst_case_nav_matches_move_test_hardcoded_values():
    """Lock in the EXACT integer values the Move test asserts.

    Pre-shock state:
      balance = 10_000_000 + 900_000_000 = 910_000_000  (seed + simulated supply)
      total_shares = 1_000_000 + 100_000_000 = 101_000_000
      worst_case_nav = 910_000_000 * 1e9 / 101_000_000 = 9_009_900_990 (truncated)

    Post-shock (drain 30% of balance):
      drain = 910_000_000 * 3000 / 10000 = 273_000_000
      balance = 910_000_000 - 273_000_000 = 637_000_000
      worst_case_nav = 637_000_000 * 1e9 / 101_000_000 = 6_306_930_693
    """
    v = VaultState.new_seeded()
    _simulate_supply_1000_dusdc(v)

    assert v.balance == 910_000_000
    assert v.total_shares == 101_000_000
    wcn_pre = v.worst_case_nav()
    assert wcn_pre == (910_000_000 * NAV_SCALE) // 101_000_000

    drain = v.balance * 3_000 // 10_000
    assert drain == 273_000_000
    v.balance -= drain
    assert v.balance == 637_000_000

    wcn_post = v.worst_case_nav()
    assert wcn_post == (637_000_000 * NAV_SCALE) // 101_000_000


def test_worst_case_nav_uses_balance_not_total_assets_at_shock():
    """Defense in depth: even with high total_assets (representing hedge cost
    basis), worst_case_nav reads ONLY from balance. This is the key insight
    of the LTV gate per Phase 2 D-14."""
    v = VaultState.new_seeded()
    _simulate_supply_1000_dusdc(v)
    # Inflate total_assets to simulate post-hedge-mint state (hedge cost basis)
    v.total_assets += 100_000_000  # 100 DUSDC hedge cost basis

    # worst_case_nav should use balance (910M), not total_assets (1010M + seed)
    wcn = v.worst_case_nav()
    nav = v.nav_per_share()
    assert wcn < nav, "worst_case_nav must be < current nav (post-hedge-mint)"
    assert wcn == (v.balance * NAV_SCALE) // v.total_shares
```
