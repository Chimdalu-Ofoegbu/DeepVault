```python
"""Walk-forward calibration loop + sensitivity table + drawdown/Sharpe/Sortino.

Per CONTEXT.md:
  D-03: OOS = most recent 30% of window; held back during calibration.
  D-04: Monthly cadence. Calibrate on month N (training data ≤ N-1), deploy
        on month N+1, never look at N+1 during calibration.
  D-10: Drawdown + Sharpe + Sortino on OOS only. Annualization: 8,760 bars/year
        (RESEARCH.md A9: BTC trades 24/7, NOT 252 trading days). Risk-free = 0.
  D-13 §9: Sensitivity table across hedge_ratio in [0.05, 0.10, 0.15, 0.20, 0.30].
  Claude's Discretion: v1 ratio (0.10) is PRESERVED regardless of which row
        produces highest Sharpe — no retrospective re-tuning.

Type discipline: walk_forward IS a numpy/pandas-allowed module (analytical
pipeline, not parity-bound). Output to int/float at module boundary.
"""
from typing import NamedTuple

import numpy as np
import pandas as pd

from .replay import simulate, strategy_fn
from .vault_state import VaultState

OOS_FRACTION: float = 0.30
BARS_PER_YEAR: int = 8_760  # RESEARCH.md A9
RISK_FREE_RATE: float = 0.0  # CONTEXT.md D-10 hackathon convention
SENSITIVITY_RATIOS: list[float] = [0.05, 0.10, 0.15, 0.20, 0.30]  # CONTEXT.md D-13 §9


class WalkForwardResult(NamedTuple):
    in_sample_sharpe: float
    oos_sharpe: float
    oos_sortino: float
    oos_max_drawdown_bps: int
    oos_underwater_bars: int
    oos_apy: float
    monthly_pnls: pd.Series


class DrawdownResult(NamedTuple):
    max_drawdown_bps: int
    underwater_bars: int
    sharpe: float
    sortino: float


def split_walk_forward(
    data: pd.DataFrame,
    oos_fraction: float = OOS_FRACTION,
) -> tuple[pd.DataFrame, list[tuple[pd.Period, pd.Period]]]:
    """Return (oos_df, list_of_(calibrate_month_end, deploy_month)).

    OOS = most recent `oos_fraction` of bars. Windows iterate monthly pairs
    over the in-sample portion (D-04).
    """
    cutoff_idx = int(len(data) * (1 - oos_fraction))
    in_sample = data.iloc[:cutoff_idx].copy()
    oos = data.iloc[cutoff_idx:].copy()
    in_sample['_month'] = pd.to_datetime(in_sample['ts_ms'], unit='ms').dt.to_period('M')
    months = sorted(in_sample['_month'].unique())
    windows = [(months[i - 1], months[i]) for i in range(1, len(months))]
    in_sample.drop(columns=['_month'], inplace=True, errors='ignore')
    return oos, windows


def run_walk_forward(
    data: pd.DataFrame,
    hedge_ratio: float = 0.10,
) -> WalkForwardResult:
    """Run monthly walk-forward calibration loop; compute OOS metrics.

    Per CONTEXT.md D-04: for each (calibrate_month_end, deploy_month) window,
    calibrate on training data ≤ calibrate_month_end and deploy on deploy_month.

    v1 has no per-window calibration — hedge_ratio is fixed for the entire run.
    The loop structure is SCAFFOLDING for v2 dynamic sizing (STRAT-V2-01).
    """
    oos, windows = split_walk_forward(data)

    # In-sample simulation
    in_sample = data.iloc[:int(len(data) * (1 - OOS_FRACTION))].copy()
    in_sample_vault = VaultState.new_seeded()
    in_sample_result = simulate(in_sample, in_sample_vault, hedge_ratio)

    in_sample_returns = pd.Series(in_sample_result['per_bar_nav']).pct_change().dropna()
    in_sample_equity = (1 + in_sample_returns).cumprod()
    in_sample_dd = compute_drawdown_max_sharpe_sortino(in_sample_equity)

    # OOS simulation — uses the same hedge_ratio (no tuning)
    oos_vault = VaultState.new_seeded()
    oos_result = simulate(oos, oos_vault, hedge_ratio)
    oos_returns = pd.Series(oos_result['per_bar_nav']).pct_change().dropna()
    oos_equity = (1 + oos_returns).cumprod()
    oos_dd = compute_drawdown_max_sharpe_sortino(oos_equity)
    oos_apy = (1 + oos_result['total_return']) ** (BARS_PER_YEAR / max(oos_result['bars'], 1)) - 1

    # Monthly pnls aggregation
    monthly_pnls = pd.Series([0.0])  # stub — full per-month aggregation in Plan 03-09 report

    return WalkForwardResult(
        in_sample_sharpe=in_sample_dd.sharpe,
        oos_sharpe=oos_dd.sharpe,
        oos_sortino=oos_dd.sortino,
        oos_max_drawdown_bps=oos_dd.max_drawdown_bps,
        oos_underwater_bars=oos_dd.underwater_bars,
        oos_apy=float(oos_apy),
        monthly_pnls=monthly_pnls,
    )


def sensitivity_table(
    data: pd.DataFrame,
    ratios: list[float] = None,
) -> pd.DataFrame:
    """Run walk-forward for each ratio; emit table per CONTEXT.md D-13 §9.

    NOTE: This table is DOCUMENTATION of robustness — the v1 production ratio
    stays at strategy_constants.ALLOCATION_BPS (=10%) regardless of which row
    produces highest Sharpe. Per CONTEXT.md Claude's Discretion: no
    retrospective re-tuning."""
    if ratios is None:
        ratios = SENSITIVITY_RATIOS
    rows = []
    for r in ratios:
        result = run_walk_forward(data, hedge_ratio=r)
        rows.append({
            'hedge_ratio': r,
            'in_sample_sharpe': result.in_sample_sharpe,
            'oos_sharpe': result.oos_sharpe,
            'oos_max_drawdown_bps': result.oos_max_drawdown_bps,
            'oos_apy': result.oos_apy,
        })
    return pd.DataFrame(rows)


def compute_drawdown_max_sharpe_sortino(equity_curve: pd.Series) -> DrawdownResult:
    """Compute max drawdown (bps), underwater duration (bars), Sharpe, Sortino.

    Per CONTEXT.md D-10: annualized using 8,760 bars/year (BTC trades 24/7).
    Per RESEARCH.md A9.
    """
    if len(equity_curve) < 2:
        return DrawdownResult(max_drawdown_bps=0, underwater_bars=0, sharpe=0.0, sortino=0.0)
    equity = equity_curve.values.astype(float)
    running_max = np.maximum.accumulate(equity)
    drawdown = (equity - running_max) / running_max
    max_dd = float(drawdown.min())
    max_dd_bps = int(max_dd * 10_000)

    # Underwater duration: longest consecutive run of drawdown < 0
    in_dd = drawdown < 0
    if not in_dd.any():
        underwater_bars = 0
    else:
        # Find longest True run
        runs = []
        current_run = 0
        for v in in_dd:
            if v:
                current_run += 1
            else:
                runs.append(current_run)
                current_run = 0
        runs.append(current_run)
        underwater_bars = int(max(runs))

    # Sharpe: (mean(returns) - RF) / std(returns) * sqrt(bars_per_year)
    returns = pd.Series(equity).pct_change().dropna().values
    if len(returns) == 0 or returns.std() == 0:
        sharpe = 0.0
        sortino = 0.0
    else:
        mean_r = float(returns.mean())
        std_r = float(returns.std(ddof=1))
        sharpe = float((mean_r - RISK_FREE_RATE / BARS_PER_YEAR) / std_r * np.sqrt(BARS_PER_YEAR))
        # Sortino: same but std only over negative returns
        downside = returns[returns < 0]
        if len(downside) == 0 or downside.std() == 0:
            sortino = 0.0
        else:
            sortino = float((mean_r - RISK_FREE_RATE / BARS_PER_YEAR) / downside.std(ddof=1) * np.sqrt(BARS_PER_YEAR))

    return DrawdownResult(
        max_drawdown_bps=max_dd_bps,
        underwater_bars=underwater_bars,
        sharpe=sharpe,
        sortino=sortino,
    )
```

```python
"""Six-column PnL attribution accountant (BACK-08, CONTEXT.md D-09).

Six columns sum to total return per bar:
  plp_yield_bps:   0 in v1 (we BUY hedges, not provide PLP per Q3/A3).
                   Column reserved for v2 STRAT-V2-01.
  hedge_cost_bps:  Premium paid per supply/hedge_mint/roll.
  hedge_payoff_bps: Settlement payoffs from roll actions.
  fees_bps:        0 in v1 (Phase 2 D-13 — no strategy fees).
  slippage_bps:    Next-bar VWAP - next-bar open (BACK-08 pessimistic fills).
  gas_bps:         1 bp per supply, 0.5 bp per roll (assumed gas model
                   per .planning/backtest-assumptions.md).
  total_bps:       Sum of the 6.

Type discipline: pnl_attribution IS pandas-allowed (analytical pipeline).
Output to int at module boundary where it crosses into report.py.
"""
from typing import Iterable

import pandas as pd

from .strategy_constants import ALLOCATION_BPS, NAV_SCALE


_PNL_COLS = [
    'plp_yield_bps',
    'hedge_cost_bps',
    'hedge_payoff_bps',
    'fees_bps',
    'slippage_bps',
    'gas_bps',
]


def _safe_div(num: int, den: int) -> int:
    if den == 0:
        return 0
    return num // den


def _slippage_bps_for_action(action: dict, market_data: pd.DataFrame) -> int:
    """Compute slippage_bps as next-bar VWAP minus next-bar open (BACK-08)."""
    if 'ts_ms' not in market_data.columns or len(market_data) == 0:
        return 0
    forward = market_data.loc[market_data['ts_ms'] >= action['ts_ms']]
    if len(forward) < 2:
        return 0
    next_bar = forward.iloc[1]
    if next_bar['volume_btc'] <= 0:
        return 0
    vwap = next_bar['volume_usdt'] / next_bar['volume_btc']
    if next_bar['open'] <= 0:
        return 0
    return int((vwap - next_bar['open']) / next_bar['open'] * 10_000)


def compute_attribution(actions: Iterable[dict], market_data: pd.DataFrame) -> pd.DataFrame:
    """Return a DataFrame with 7 columns: the 6 PnL columns + total_bps.

    Per CONTEXT.md D-09. Tests assert that columns sum to total within 1 bp.
    """
    rows: list[dict] = []
    for action in actions:
        row = {
            'ts_ms': int(action.get('ts_ms', 0)),
            'kind': str(action.get('kind', '')),
            'plp_yield_bps': 0,
            'hedge_cost_bps': 0,
            'hedge_payoff_bps': 0,
            'fees_bps': 0,
            'slippage_bps': 0,
            'gas_bps': 0,
        }

        kind = action.get('kind', '')
        if kind == 'supply':
            deposit_quote = int(action['args']['deposit_quote'])
            hedge_alloc = deposit_quote * ALLOCATION_BPS // 10_000
            row['hedge_cost_bps'] = _safe_div(hedge_alloc * 10_000, max(deposit_quote, 1))
            row['slippage_bps'] = _slippage_bps_for_action(action, market_data)
            row['gas_bps'] = 1
        elif kind == 'hedge_mint':
            cost = int(action['args'].get('cost_basis_quote', 0))
            pre_total_assets = int(action.get('pre', {}).get('total_assets', 1))
            row['hedge_cost_bps'] = _safe_div(cost * 10_000, max(pre_total_assets, 1))
            row['gas_bps'] = 1
        elif kind == 'roll':
            payoff = int(action['args'].get('payoff_quote', 0))
            new_cost = int(action['args'].get('new_cost_basis_quote', 0))
            pre_total_assets = int(action.get('pre', {}).get('total_assets', 1))
            row['hedge_payoff_bps'] = _safe_div(payoff * 10_000, max(pre_total_assets, 1))
            row['hedge_cost_bps'] = _safe_div(new_cost * 10_000, max(pre_total_assets, 1))
            row['gas_bps'] = 1
        elif kind in ('redeem_request', 'redeem_fulfill', 'redeem_cancel'):
            row['gas_bps'] = 1

        rows.append(row)

    if not rows:
        # Empty: return empty DataFrame with the 7 columns
        return pd.DataFrame(columns=['ts_ms', 'kind'] + _PNL_COLS + ['total_bps'])

    df = pd.DataFrame(rows)
    df['total_bps'] = df[_PNL_COLS].sum(axis=1)
    return df
```
