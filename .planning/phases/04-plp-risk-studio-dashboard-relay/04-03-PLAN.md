    // Intl.NumberFormat helpers for u64 bigint display.
    // Pitfall 8: parse wire-format strings via BigInt; cast to Number only at the
    // display boundary, divided by the scale factor.
    import { STRATEGY_CONSTANTS } from './strategy_constants';

    const NAV_FMT = new Intl.NumberFormat('en-US', {
      minimumFractionDigits: 2, maximumFractionDigits: 6, useGrouping: true,
    });
    const DUSDC_DECIMALS = 6n;
    const DUSDC_DIVISOR = 1000000n; // 10^6
    const NAV_SCALE = STRATEGY_CONSTANTS.NAV_SCALE; // 1_000_000_000n per Phase 2 D-15

    function bigintToDisplay(value: bigint, divisor: bigint, fmt: Intl.NumberFormat): string {
      const whole = value / divisor;
      const frac = value % divisor;
      const fracPadded = frac.toString().padStart(divisor.toString().length - 1, '0');
      const scaled = Number(`${whole}.${fracPadded}`);
      return fmt.format(scaled);
    }

    /** DUSDC at 6 decimals. Input is bigint u64 from wire. */
    export function formatDusdc(quoteMicro: bigint): string {
      return bigintToDisplay(quoteMicro, DUSDC_DIVISOR, NAV_FMT);
    }

    /** NAV at NAV_SCALE 10^9. */
    export function formatNav(navScaled: bigint): string {
      return bigintToDisplay(navScaled, NAV_SCALE, NAV_FMT);
    }

    /** Share count (raw u64; SHARE coin has its own decimals). */
    export function formatShares(shares: bigint): string {
      return NAV_FMT.format(Number(shares));
    }

    /** Basis points -> "1500 bps (15.00%)" per UI-SPEC tone callout. */
    export function formatBps(bps: bigint | number): string {
      const n = typeof bps === 'bigint' ? Number(bps) : bps;
      return `${n} bps (${(n / 100).toFixed(2)}%)`;
    }

    /** Truncate Sui tx digest as 0x1234…abcd per UI-SPEC copywriting contract. */
    export function truncateDigest(digest: string): string {
      if (digest.length <= 14) return digest;
      const prefix = digest.startsWith('0x') ? digest.slice(0, 6) : `0x${digest.slice(0, 4)}`;
      const suffix = digest.slice(-4);
      return `${prefix}…${suffix}`;
    }
    ```

    import { useEffect, useState } from 'react';
    import type { Staleness } from '@/lib/types';

    export function useStaleness(lastUpdatedMs: number | null): Staleness {
      const [now, setNow] = useState(() => Date.now());
      useEffect(() => {
        const id = setInterval(() => setNow(Date.now()), 1000);
        return () => clearInterval(id);
      }, []);
      if (lastUpdatedMs == null) return 'stale';
      const delta = now - lastUpdatedMs;
      if (delta < 30_000) return 'fresh';
      if (delta < 60_000) return 'warning';
      return 'stale';
    }
    ```

    // Worst-case staleness across panels = max age of all `last_updated_ms` fields in the snapshot.
    import { StalenessPill } from '@/components/primitives/StalenessPill';
    import type { FullSnapshot } from '@/lib/types';

    type Props = { snapshot: FullSnapshot | null };
    export function GlobalStalenessPill({ snapshot }: Props) {
      if (!snapshot) return <StalenessPill lastUpdatedMs={null} compact />;
      const oracleMin = snapshot.oracles.length === 0 ? null
        : Math.min(...snapshot.oracles.map((o) => Number(o.last_updated_ms)));
      const vaultTs = snapshot.vault ? Number(snapshot.vault.last_updated_ms) : null;
      const oldest = [oracleMin, vaultTs].filter((x): x is number => x != null).reduce((a, b) => Math.min(a, b), Date.now());
      return <StalenessPill lastUpdatedMs={oldest} />;
    }
