    ```typescript
    import { useMemo } from 'react';
    import type { FullSnapshot, SurfaceSnapshot } from '@/lib/types';
    import type { SVIParams } from '@/lib/svi';

    export type SurfaceView = {
      raw: SurfaceSnapshot;
      svi: SVIParams;
      lastUpdatedMs: number;
    } | null;

    export function useSurfaceSnapshot(snapshot: FullSnapshot | null, oracleId?: string): SurfaceView {
      return useMemo(() => {
        if (!snapshot || snapshot.oracles.length === 0) return null;
        const o = oracleId
          ? snapshot.oracles.find((x) => x.oracle_id === oracleId)
          : snapshot.oracles[0]; // v1 single BTC oracle
        if (!o) return null;
        const svi: SVIParams = {
          a: BigInt(o.a),
          b: BigInt(o.b),
          rho: BigInt(o.rho_signed),
          m: BigInt(o.m_signed),
          sigma: BigInt(o.sigma),
        };
        return { raw: o, svi, lastUpdatedMs: Number(o.last_updated_ms) };
      }, [snapshot, oracleId]);
    }
    ```

    ```tsx
    import { Collapsible, CollapsibleContent, CollapsibleTrigger } from '@/components/ui/collapsible';

    import { ChevronDown, ChevronUp } from 'lucide-react';
    import { checkArb } from '@/lib/arb_checker';
    import type { SurfaceView } from '@/hooks/useSurfaceSnapshot';

    const STALE_THRESHOLD_MS = 5 * 60 * 1000; // UI-SPEC special case

    type Props = { surface: SurfaceView };

    type Status = 'green' | 'red' | 'stale';

    export function ArbCheckerPanel({ surface }: Props) {
      const surfaceAgeMs = surface ? Date.now() - surface.lastUpdatedMs : Infinity;
      const isStale = surfaceAgeMs > STALE_THRESHOLD_MS;
      const arb = useMemo(() => surface ? checkArb(surface.svi) : null, [surface]);
      const status: Status = isStale ? 'stale' : (!arb || !arb.paramsValid || arb.minGk < 0n || !arb.calendarPass) ? 'red' : 'green';

      // gK is bigint[] at FLOAT_SCALING 1e9 (length === 200).
      // Convert to Number for Recharts at the boundary — acceptable per arb_checker.ts:14-16
      // (arb_checker is visualization-bound, NOT parity-bound).
      const chartData = useMemo(() => {
        if (!arb || arb.gK.length === 0) return [];
        return arb.gK.map((g, i) => ({ k: i - 100, g: Number(g) / 1e9 }));
      }, [arb]);

      const [open, setOpen] = useState(false);
      const autoExpand = status === 'red';
      const expanded = autoExpand || open;

      return (
        <Card>
          <CardHeader className="flex flex-row items-center justify-between">
            <div>
              <CardTitle>Arbitrage check</CardTitle>
              <CardDescription>g(k) curve (200-point grid). Status flips RED when min g(k) &lt; 0 or calendar arbitrage detected.</CardDescription>
            </div>
            <div className="flex items-center gap-2">
              {status === 'green' && <Badge className="bg-emerald-500/15 text-emerald-300 border border-emerald-500/30 text-[11px] uppercase tracking-wider font-semibold">GREEN</Badge>}
              {status === 'red' && <Badge className="bg-rose-600/15 text-rose-300 border border-rose-600/30 text-[11px] uppercase tracking-wider font-semibold">RED</Badge>}
              {status === 'stale' && <Badge className="bg-slate-700/30 text-slate-300 border border-slate-600 text-[11px] uppercase tracking-wider font-semibold">STALE — CANNOT VERIFY</Badge>}
              {surface && <StalenessPill lastUpdatedMs={surface.lastUpdatedMs} compact />}
            </div>
          </CardHeader>
          <CardContent>
            {status === 'stale' ? (
              <p className="text-sm text-slate-400">Arbitrage check requires SVI fresher than 5 minutes.</p>
            ) : !arb ? (
              <p className="text-sm text-slate-400">Waiting for first SVI update.</p>
            ) : (
              <div>
                <div className="flex items-center justify-between text-sm text-slate-400">
                  <span>min g(k) = <span className="font-mono tabular-nums text-slate-200">{(Number(arb.minGk) / 1e9).toFixed(6)}</span></span>
                  <Collapsible open={expanded} onOpenChange={setOpen}>
                    <CollapsibleTrigger asChild>
                      <Button variant="ghost" size="sm">
                        {expanded ? <><ChevronUp className="h-4 w-4 mr-1" /> Hide g(k) curve</> : <><ChevronDown className="h-4 w-4 mr-1" /> View g(k) curve</>}

                          <XAxis dataKey="k" tick={{ fill: '#94a3b8', fontSize: 11 }} label={{ value: 'log-strike k', position: 'insideBottom', offset: -2, fill: '#94a3b8' }} />

    }
    ```

    ```tsx
    import Plot from 'react-plotly.js';
    import { useMemo, useState, useEffect } from 'react';
    import type { Data, Layout, Config } from 'plotly.js';
    import { Card, CardHeader, CardTitle, CardDescription, CardContent } from '@/components/ui/card';
    import { Skeleton } from '@/components/ui/skeleton';
    import { StalenessPill } from '@/components/primitives/StalenessPill';
    import { totalVariance, type SVIParams } from '@/lib/svi';
    import type { SurfaceView } from '@/hooks/useSurfaceSnapshot';

    const K_GRID = 50;
    const K_MIN = -2.0;
    const K_MAX = 2.0;
    const FLOAT_SCALING = 1_000_000_000n;
    // v1 single BTC oracle (CONTEXT.md A9); show a single-tenor strip by default.
    // The surface still renders meaningfully — Y axis spans [now+7d, now+14d, now+30d].
    const DEFAULT_TENORS_YEARS = [7 / 365, 14 / 365, 30 / 365];

    type Props = { surface: SurfaceView; tenors?: number[] };

    export function SurfacePanel({ surface, tenors = DEFAULT_TENORS_YEARS }: Props) {
      const grid = useMemo(() => {
        if (!surface) return null;
        const ks: number[] = [];
        for (let i = 0; i < K_GRID; i++) {
          ks.push(K_MIN + ((K_MAX - K_MIN) * i) / (K_GRID - 1));
        }
        // For v1 single-oracle, all tenors share the same SVI params (the OracleSVI
        // emits one surface per oracle; multi-tenor lifting is STRAT-V2-03). Render
        // the same surface across the Y axis as a smooth ribbon. When multi-tenor
        // surfaces ship, the executor swaps `surface.svi` for a tenor-indexed lookup.
        const svi: SVIParams = surface.svi;
        const z: number[][] = [];
        for (let j = 0; j < tenors.length; j++) {
          const row: number[] = [];
          for (let i = 0; i < K_GRID; i++) {
            const kFloat = ks[i];
            const kBig = BigInt(Math.trunc(kFloat * Number(FLOAT_SCALING)));
            const w = totalVariance(svi, kBig);
            row.push(Number(w) / Number(FLOAT_SCALING));
          }
          z.push(row);
        }
        return { x: ks, y: tenors, z };
      }, [surface, tenors]);

      const data: Data[] = useMemo(() => grid ? [{
        type: 'surface',
        x: grid.x,
        y: grid.y,
        z: grid.z,
        colorscale: 'Viridis',
        showscale: false,
        contours: { z: { show: true, usecolormap: true, project: { z: true } } },
      } as Data] : [], [grid]);

      const layout: Partial<Layout> = useMemo(() => ({
        scene: {
          xaxis: { title: { text: 'log-strike k' }, color: '#94a3b8' },
          yaxis: { title: { text: 'tenor T (years)' }, color: '#94a3b8' },
          zaxis: { title: { text: 'total variance w(k,T)' }, color: '#94a3b8' },
          camera: { eye: { x: 1.3, y: 1.3, z: 0.9 } },
        },
        margin: { t: 0, r: 0, b: 0, l: 0 },
        paper_bgcolor: 'rgba(0,0,0,0)',
        plot_bgcolor: 'rgba(0,0,0,0)',
        font: { color: '#94a3b8', family: 'Inter' },
        autosize: true,
      }), []);

      const config: Partial<Config> = useMemo(() => ({
        scrollZoom: false,
        displayModeBar: false,
        responsive: true,
      }), []);

      const [revision, setRevision] = useState(0);
      useEffect(() => {
        if (surface) setRevision((r) => r + 1);
      }, [surface?.raw.timestamp_ms]);

      return (
        <Card>
          <CardHeader className="flex flex-row items-start justify-between">
            <div>
              <CardTitle>BTC volatility surface (live)</CardTitle>
              <CardDescription>50×50 grid: log-strike k × tenor T × total variance w(k,T). Re-renders on every OracleSVIUpdated event.</CardDescription>
            </div>
            {surface && <StalenessPill lastUpdatedMs={surface.lastUpdatedMs} />}
          </CardHeader>
          <CardContent>
            {!surface || !grid ? (
              <div className="space-y-4">
                <Skeleton className="h-[600px] w-full" />
                <p className="text-sm text-slate-400">Waiting for first SVI update — the relay is connected and listening.</p>
              </div>
            ) : (
              <div data-testid="surface-plot" style={{ width: '100%', height: 600 }}>
                <Plot
                  data={data}
                  layout={layout}
                  config={config}
                  revision={revision}
                  style={{ width: '100%', height: '100%' }}
                  useResizeHandler
                />
              </div>
            )}
          </CardContent>
        </Card>
      );
    }
    ```
