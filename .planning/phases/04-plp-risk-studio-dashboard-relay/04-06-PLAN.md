    Create `dashboard/src/lib/whatIf.ts`:
    ```typescript
    // What-if simulator pure-compute layer (D-09 sub-100ms requirement).
    // All math at bigint at FLOAT_SCALING 1e9; only the slider/UI layer touches Number.
    // Re-uses Phase 1 SVI lib unchanged (forbidden-token grep enforces no Number/Math.X here).
    import { binaryPrice, type SVIParams } from './svi';
    import { STRATEGY_CONSTANTS } from './strategy_constants';

    const FLOAT_SCALING = STRATEGY_CONSTANTS.NAV_SCALE; // 1_000_000_000n

    /** Parallel theta shock: scale `a` (variance level) by (1 + theta_shock_bps/10000).
     *  Other params (b, rho, m, sigma) unchanged. D-07 approximation for v1.
     *  Clamps `a` at 0n to preserve non-negativity invariant. */
    export function shockSviParallel(svi: SVIParams, thetaShockBps: number): SVIParams {
      if (thetaShockBps === 0) return svi;
      const bps = BigInt(thetaShockBps);
      const delta = (svi.a * bps) / 10000n;
      const a = svi.a + delta;
      return { ...svi, a: a < 0n ? 0n : a };
    }

    /** Spot shock on forward price: forward * (1 + spot_shock_bps/10000). Clamps at 1n.
     *  D-08 — ±5σ at fallback 20% sigma_F. */
    export function shockedForward(forward: bigint, spotShockBps: number): bigint {
      if (spotShockBps === 0) return forward;
      const bps = BigInt(spotShockBps);
      const delta = (forward * bps) / 10000n;
      const out = forward + delta;
      return out < 1n ? 1n : out;
    }

    export type ShockedQuote = {
      hedgeKey: string;
      baselinePrice: bigint;   // binaryPrice with shock=0
      shockedPrice: bigint;
      baselineNotional: bigint;
      pnlQuote: bigint;        // (shockedPrice * notional - baselinePrice * notional) / FLOAT_SCALING
    };

    /** Per-hedge shocked PnL. The signature of binaryPrice is verified against
     *  dashboard/src/lib/svi.ts at task time; if it requires more args than below,
     *  the executor adapts but keeps `svi`, `forward`, `strike`, `tenor_seconds`
     *  as primary inputs. The mapping from Hedge to binaryPrice args:
     *    - svi: shockSviParallel(surface.svi, thetaShockBps)
     *    - forward: shockedForward(hedge.forward (or surface.forward), spotShockBps)
     *      [if Hedge does not carry forward, fall back to a default forward derived
     *       from surface; v1 single-oracle case ships a synthesized forward]
     *    - strike: hedge.strike
     *    - tenor_seconds: (hedge.expiryMs - Date.now()) / 1000n */
    export function shockedPnL(args: {
      hedgeKey: string;
      svi: SVIParams;
      forward: bigint;
      strike: bigint;
      notionalQuote: bigint;
      tenorSeconds: bigint;
      thetaShockBps: number;
      spotShockBps: number;
    }): ShockedQuote {
      const baselineSvi = args.svi;
      const shockedSvi = shockSviParallel(args.svi, args.thetaShockBps);
      const baselineFwd = args.forward;
      const shockedFwd = shockedForward(args.forward, args.spotShockBps);
      // Executor: invoke binaryPrice with whatever signature dashboard/src/lib/svi.ts
      // actually exports. The wrapping shape stays the same.
      const baselinePrice = binaryPrice({
        svi: baselineSvi,
        forward: baselineFwd,
        strike: args.strike,
        tenor_seconds: args.tenorSeconds,
      } as any);
      const shockedPrice = binaryPrice({
        svi: shockedSvi,
        forward: shockedFwd,
        strike: args.strike,
        tenor_seconds: args.tenorSeconds,
      } as any);
      const pnlQuote = ((shockedPrice - baselinePrice) * args.notionalQuote) / FLOAT_SCALING;
      return {
        hedgeKey: args.hedgeKey,
        baselinePrice,
        shockedPrice,
        baselineNotional: args.notionalQuote,
        pnlQuote,
      };
    }
    ```

    Then create `dashboard/src/hooks/useSigmaEstimates.ts` (NOTE the explicit { value, isBootstrap, observationCount } shape — D-08 partial-delivery contract):
    ```typescript
    import { useMemo } from 'react';
    import type { FullSnapshot } from '@/lib/types';
    import {
      BOOTSTRAP_SIGMA_THETA_PCT,
      BOOTSTRAP_SIGMA_SPOT_PCT,
      BOOTSTRAP_MIN_OBSERVATIONS,
      SIGMA_ROLLING_WINDOW_MS,
      SIGMA_THETA_PCT_CAP,
    } from '@/lib/dashboard_constants';

    export type SigmaEstimates = {
      sigmaThetaPct: number;
      sigmaSpotPct: number;
      /** True when EITHER leg is on its bootstrap fallback. In v1 the spot leg
       *  is ALWAYS bootstrapped (OracleSVIUpdated does not carry forward price),
       *  so isBootstrap is always true. Caller renders an amber bootstrap badge. */
      isBootstrap: boolean;
      /** Theta-leg observation count from the rolling 30-day window. 0 when
       *  snapshot is null. Surface this in the UI alongside isBootstrap so the
       *  user can see "using bootstrap because only 3 observations" vs
       *  "using bootstrap because spot stream missing". */
      observationCount: number;
      /** True when the theta leg is on bootstrap (subset of isBootstrap; lets the
       *  caller distinguish theta-vs-spot bootstrap reasons in the UI copy). */
      isThetaBootstrap: boolean;
    };

    export function useSigmaEstimates(snapshot: FullSnapshot | null): SigmaEstimates {
      return useMemo(() => {
        if (!snapshot) {
          return {
            sigmaThetaPct: BOOTSTRAP_SIGMA_THETA_PCT,
            sigmaSpotPct: BOOTSTRAP_SIGMA_SPOT_PCT,
            isBootstrap: true,
            isThetaBootstrap: true,
            observationCount: 0,
          };
        }
        const cutoff = Date.now() - SIGMA_ROLLING_WINDOW_MS;
        const oracleEvents = snapshot.ring_buffer.filter(
          (e) => e.name === 'OracleSVIUpdated' && Number(e.ts_ms) >= cutoff,
        );
        if (oracleEvents.length < BOOTSTRAP_MIN_OBSERVATIONS) {
          return {
            sigmaThetaPct: BOOTSTRAP_SIGMA_THETA_PCT,
            sigmaSpotPct: BOOTSTRAP_SIGMA_SPOT_PCT,
            isBootstrap: true,
            isThetaBootstrap: true,
            observationCount: oracleEvents.length,
          };
        }
        // Theta-leg: live rolling stdev of \`a\` across observations.
        const aValues = oracleEvents.map((e) => Number((e.data as any).a ?? 0) / 1e9);
        const mean = aValues.reduce((a, b) => a + b, 0) / aValues.length;
        const variance =
          aValues.reduce((s, v) => s + (v - mean) ** 2, 0) / (aValues.length - 1);
        const stdev = Math.sqrt(variance);
        const sigmaThetaPct =
          mean === 0
            ? BOOTSTRAP_SIGMA_THETA_PCT
            : Math.min(SIGMA_THETA_PCT_CAP, (stdev / mean) * 100);
        // Spot leg: D-08 partial-delivery — OracleSVIUpdated does NOT carry forward
        // price, so rolling 30-day stdev of forward price is unavailable from the
        // event stream alone. Always return bootstrap value with isBootstrap=true.
        // Document this in 04-06-SUMMARY.md.
        return {
          sigmaThetaPct,
          sigmaSpotPct: BOOTSTRAP_SIGMA_SPOT_PCT,
          isBootstrap: true, // spot leg forces this even when theta is live
          isThetaBootstrap: false,
          observationCount: oracleEvents.length,
        };
      }, [snapshot]);
    }
    ```

    import { Card, CardHeader, CardTitle, CardDescription, CardContent } from '@/components/ui/card';
    import { Button } from '@/components/ui/button';
    import { Slider } from '@/components/ui/slider';
    import { Badge } from '@/components/ui/badge';

    import { shockedPnL } from '@/lib/whatIf';
    import { formatDusdc } from '@/lib/format';
    import { FALLBACK_FORWARD_PRICE_DUSDC } from '@/lib/dashboard_constants';
    import type { Hedge } from '@/hooks/useExposure';
    import type { SurfaceView } from '@/hooks/useSurfaceSnapshot';
    import type { SigmaEstimates } from '@/hooks/useSigmaEstimates';

    type Props = {
      hedges: Hedge[];
      surface: SurfaceView;
      sigma: SigmaEstimates;
      // v1 single-oracle: forward price source. When undefined, FALLBACK_FORWARD_PRICE_DUSDC
      // is used AND an amber "synthetic forward" badge is rendered in CardDescription so
      // the user knows the PnL output is anchored to a synthetic forward, not live spot.
      forwardPrice?: bigint;
    };

    export function WhatIfSimulator({ hedges, surface, sigma, forwardPrice }: Props) {
      const [thetaShockBps, setThetaShockBps] = useState(0);
      const [spotShockBps, setSpotShockBps] = useState(0);
      const cardRef = useRef<HTMLDivElement>(null);

      // Slider ranges (UI-SPEC §What-if simulator)
      const SPOT_MAX = 50_000;  // ±5σ at 100% sigma; with 20% bootstrap = ±5σ → ±100%
      const THETA_MAX = 4_000;  // ±2σ at 100% sigma_theta; with 20% bootstrap = ±40%

      const isSyntheticForward = forwardPrice === undefined;
      const fwd = forwardPrice ?? FALLBACK_FORWARD_PRICE_DUSDC;
      const now = Date.now();

      const quotes = useMemo(() => {
        if (!surface || hedges.length === 0) return [];
        return hedges.map((h) => shockedPnL({
          hedgeKey: h.marketKey,
          svi: surface.svi,
          forward: fwd,
          strike: h.strike,
          notionalQuote: h.notionalQuote,
          tenorSeconds: BigInt(Math.max(1, Math.floor((h.expiryMs - now) / 1000))),
          thetaShockBps,
          spotShockBps,
        }));
      }, [hedges, surface, fwd, now, thetaShockBps, spotShockBps]);

      const totalPnl = useMemo(
        () => quotes.reduce((acc, q) => acc + q.pnlQuote, 0n),
        [quotes],
      );

      const chartData = useMemo(() => {
        const rows = quotes.map((q) => ({
          name: hedges.find((h) => h.marketKey === q.hedgeKey)?.strikeDisplay ?? q.hedgeKey,
          pnl: Number(q.pnlQuote) / 1e6, // DUSDC display
        }));
        rows.push({ name: 'Total', pnl: Number(totalPnl) / 1e6 });
        return rows;
      }, [quotes, hedges, totalPnl]);

      const onReset = useCallback(() => {
        setThetaShockBps(0);
        setSpotShockBps(0);
      }, []);

      const onKeyDown = useCallback((e: React.KeyboardEvent) => {
        if (e.key === 'Escape') onReset();
      }, [onReset]);

      if (hedges.length === 0) {
        return (
          <Card>
            <CardHeader>
              <CardTitle>What-if simulator</CardTitle>
              <CardDescription>Joint spot + vol shocks on your open hedges.</CardDescription>
            </CardHeader>
            <CardContent>
              <p className="font-semibold text-slate-200">Connect wallet to simulate</p>
              <p className="mt-1 text-sm text-slate-400">Sliders shock your open hedges by spot ±5σ and vol ±2σ. PnL is recomputed in your browser.</p>
            </CardContent>
          </Card>
        );
      }

      return (
        <Card>
          <CardHeader className="flex flex-row items-start justify-between">
            <div>
              <CardTitle>What-if simulator</CardTitle>
              <CardDescription>
                Joint spot + vol shocks on your open hedges. Client-side compute; sub-100ms response per slider tick.
                {(sigma.isBootstrap || isSyntheticForward) && (
                  <span className="mt-2 flex flex-wrap gap-2">
                    {sigma.isBootstrap && (
                      <Badge variant="amber" title={
                        sigma.isThetaBootstrap
                          ? `Bootstrap σ: spot-leg σ uses fallback (OracleSVIUpdated does not carry forward price) AND theta-leg σ uses fallback (${sigma.observationCount} observations < 7 required).`
                          : `Bootstrap σ: spot-leg σ uses fallback (OracleSVIUpdated does not carry forward price). Theta-leg σ is live (${sigma.observationCount} observations).`
                      }>
                        Bootstrap σ
                      </Badge>
                    )}
                    {isSyntheticForward && (
                      <Badge variant="amber">Using synthetic forward — connect oracle for live pricing</Badge>
                    )}
                  </span>
                )}
              </CardDescription>
            </div>
            <Button variant="ghost" size="sm" onClick={onReset} aria-label="Reset sliders">Reset to current</Button>
          </CardHeader>
          <CardContent>
            <div ref={cardRef} onKeyDown={onKeyDown} tabIndex={-1} className="space-y-6">
              <div>
                <label className="flex justify-between text-sm text-slate-400">
                  <span>Spot shock (forward price)</span>
                  <span className="font-mono tabular-nums text-slate-200">{(spotShockBps / 100).toFixed(2)}%</span>
                </label>
                <Slider min={-SPOT_MAX} max={SPOT_MAX} step={100} value={[spotShockBps]} onValueChange={(v) => setSpotShockBps(v[0] ?? 0)} className="mt-2" />
                <p className="mt-1 text-xs text-slate-500">±5σ ≈ ±{(sigma.sigmaSpotPct * 5).toFixed(1)}% {sigma.isBootstrap && '(bootstrap fallback: <7d history; using 20% σ)'}</p>
              </div>
              <div>
                <label className="flex justify-between text-sm text-slate-400">
                  <span>Vol shock (parallel θ shift)</span>
                  <span className="font-mono tabular-nums text-slate-200">{(thetaShockBps / 100).toFixed(2)}%</span>
                </label>
                <Slider min={-THETA_MAX} max={THETA_MAX} step={50} value={[thetaShockBps]} onValueChange={(v) => setThetaShockBps(v[0] ?? 0)} className="mt-2" />
                <p className="mt-1 text-xs text-slate-500">±2σ ≈ ±{(sigma.sigmaThetaPct * 2).toFixed(1)}% {sigma.isBootstrap && '(bootstrap fallback: <7d θ history; using 20% σ_θ)'}</p>
              </div>
              <div data-testid="pnl-chart" style={{ width: '100%', height: 220 }}>
                <ResponsiveContainer width="100%" height="100%">
                  <BarChart data={chartData}>
                    <XAxis dataKey="name" tick={{ fill: '#94a3b8', fontSize: 11 }} />
                    <YAxis tick={{ fill: '#94a3b8', fontSize: 11 }} />
                    <RcTooltip contentStyle={{ background: '#0f172a', border: '1px solid #334155', borderRadius: 6 }} formatter={(value: number) => [`${value.toFixed(2)} DUSDC`, 'PnL']} />
                    <Bar dataKey="pnl" isAnimationActive={false}>
                      {chartData.map((row, i) => (
                        <Cell key={i} fill={row.pnl >= 0 ? '#06b6d4' : '#e11d48'} />
                      ))}
                    </Bar>
                  </BarChart>
                </ResponsiveContainer>
              </div>
              <p className="text-right text-sm text-slate-400">
                Total PnL: <NumericValue className={totalPnl >= 0n ? 'text-cyan-300' : 'text-rose-300'}>{formatDusdc(totalPnl)} DUSDC</NumericValue>
              </p>
            </div>
          </CardContent>
        </Card>
      );
    }
