    // PTB construction for vault::supply, vault::redeem (request/fulfill/cancel).
    // Analog: scripts/e2e-vault-cycle.ts:180-329, scripts/two-protocol-ptb-demo.ts:286-457.
    // Shared-object-ref discipline: vault + predict_manager use sharedObjectRef;
    // oracle_svi + predict_top_level use tx.object; Clock 0x6 uses tx.object('0x6').
    import { Transaction } from '@mysten/sui/transactions';
    import type { DeployJson } from './ptbDeploy';

    export function buildSupplyTx(args: {
      deploy: DeployJson;
      depositCoinId: string;       // pre-existing Coin<DUSDC> objectId owned by sender
      depositAmountMicro: bigint;  // amount in DUSDC micro-units (6 decimals)
    }): Transaction {
      const tx = new Transaction();
      const [deposit] = tx.splitCoins(tx.object(args.depositCoinId), [
        tx.pure.u64(args.depositAmountMicro),
      ]);
      // Arg order MUST match contracts/sources/supply.move:61.
      // Analog (e2e-vault-cycle.ts:198-206): vault, predict_top_level, predict_manager, oracle_svi, deposit, clock
      tx.moveCall({
        target: `${args.deploy.package_id}::supply::supply`,
        typeArguments: [args.deploy.dusdc_type_tag],
        arguments: [
          tx.sharedObjectRef({
            objectId: args.deploy.vault_id,
            mutable: true,
            initialSharedVersion: args.deploy.vault_initial_shared_version,
          }),
          tx.object(args.deploy.predict_top_level_id),
          tx.sharedObjectRef({
            objectId: args.deploy.predict_manager_id,
            mutable: true,
            initialSharedVersion: args.deploy.predict_manager_initial_shared_version,
          }),
          tx.object(args.deploy.oracle_svi_id ?? ''),
          deposit,
          tx.object('0x6'),
        ],
      });
      return tx;
    }

    export function buildRedeemRequestTx(args: {
      deploy: DeployJson;
      shareCoinId: string;
    }): Transaction {
      const tx = new Transaction();
      tx.moveCall({
        target: `${args.deploy.package_id}::redeem::redeem_request`,
        typeArguments: [args.deploy.dusdc_type_tag],
        arguments: [
          tx.sharedObjectRef({
            objectId: args.deploy.vault_id,
            mutable: true,
            initialSharedVersion: args.deploy.vault_initial_shared_version,
          }),
          tx.object(args.shareCoinId),
          tx.object('0x6'),
        ],
      });
      return tx;
    }

    export function buildRedeemFulfillTx(args: { deploy: DeployJson }): Transaction {
      const tx = new Transaction();
      tx.moveCall({
        target: `${args.deploy.package_id}::redeem::redeem_fulfill`,
        typeArguments: [args.deploy.dusdc_type_tag],
        arguments: [
          tx.sharedObjectRef({
            objectId: args.deploy.vault_id,
            mutable: true,
            initialSharedVersion: args.deploy.vault_initial_shared_version,
          }),
          tx.object('0x6'),
        ],
      });
      return tx;
    }

    export function buildRedeemCancelTx(args: { deploy: DeployJson }): Transaction {
      const tx = new Transaction();
      tx.moveCall({
        target: `${args.deploy.package_id}::redeem::redeem_cancel`,

    }

    import { useMemo } from 'react';
    import { useCurrentAccount } from '@mysten/dapp-kit';
    import type { FullSnapshot, RingEvent } from '@/lib/types';

    export type Position = {
      eventTs: number;
      depositQuote: bigint;
      sharesMinted: bigint;
      hedgeCost: bigint;        // sum of premium_quote for HedgeMinted events around supply timestamp
      hedgePayoff: bigint;      // HedgeUnwound payout_quote
      plpYield: bigint;         // best-effort v1: 0n (Phase 1 backtest covers attribution; v1 dashboard shows split when events provide it)
      netQuote: bigint;
    };

    export function usePositions(snapshot: FullSnapshot | null): Position[] {
      const account = useCurrentAccount();
      return useMemo(() => {
        if (!snapshot || !account) return [];
        const positions: Position[] = [];
        for (const e of snapshot.ring_buffer) {
          if (!e.name.endsWith('::supply::Supplied')) continue;
          const d: any = e.data;
          if (d.supplier !== account.address) continue;
          const depositQuote = BigInt(d.deposit_quote ?? '0');
          const sharesMinted = BigInt(d.shares_minted ?? '0');
          // Best-effort: hedge cost approximated by next HedgeMinted with same vault_id within 1 minute of this Supplied
          const ts = Number(e.ts_ms);
          const hedgeMinted = snapshot.ring_buffer.find(
            (x) => x.name.endsWith('::rebalance::HedgeMinted')
                && Number(x.ts_ms) >= ts
                && Number(x.ts_ms) <= ts + 60_000,
          );
          const hedgeCost = hedgeMinted ? BigInt((hedgeMinted.data as any).premium_quote ?? '0') : 0n;
          // hedgePayoff requires HedgeUnwound match by oracle/strike/expiry → best-effort v1
          positions.push({
            eventTs: ts,
            depositQuote,
            sharesMinted,
            hedgeCost,
            hedgePayoff: 0n,
            plpYield: 0n,
            netQuote: -hedgeCost,  // best-effort: net = -hedge cost until realization
          });
        }
        return positions;
      }, [snapshot, account]);
    }

    // Lazy-init pattern: per Phase 2 redeem.move::get_or_init_user_bucket, the bucket
    // does not exist until the user's first redeem_request. Return null gracefully.
    export function useBucketState() {
      const account = useCurrentAccount();
      const client = useSuiClient();
      const { DEPLOY, isDeployed } = useDeploy(); // Plan 07 Task 1 ptbDeploy export
      const enabled = !!account && isDeployed();
      const q = useQuery({
        queryKey: ['user-bucket', account?.address ?? null, DEPLOY.vault_id],
        queryFn: async (): Promise<BucketView> => {
          if (!account) return null;
          // First read vault to extract rate_limiters Table parent id.
          const vaultObj = await client.getObject({ id: DEPLOY.vault_id, options: { showContent: true } });
          const fields = vaultObj.data?.content?.dataType === 'moveObject'
            ? (vaultObj.data.content.fields as Record<string, any>)
            : {};
          const rateLimitersParent = fields.rate_limiters?.fields?.id?.id ?? fields.rate_limiters?.id;
          if (!rateLimitersParent) return null;
          try {
            const dyn = await client.getDynamicFieldObject({
              parentId: rateLimitersParent,
              name: { type: 'address', value: account.address },
            });
            const rl = dyn.data?.content?.dataType === 'moveObject'
              ? (dyn.data.content.fields as Record<string, any>)
              : null;
            if (!rl) return null;
            const availableMicro = BigInt(rl.available_micro ?? rl.value?.fields?.current_available ?? '0');
            const capacityMicro = BigInt(rl.capacity ?? rl.value?.fields?.capacity ?? '0');
            const refillRatePerMs = BigInt(rl.refill_rate_per_ms ?? rl.value?.fields?.refill_rate_per_ms ?? '0');
            const utilizationPct = capacityMicro === 0n ? 0 : Number((availableMicro * 100n) / capacityMicro);
            return { availableMicro, capacityMicro, refillRatePerMs, utilizationPct };
          } catch {
            // Bucket not yet lazy-initialized — first redeem_request seeds it
            return null;
          }
        },
        enabled,
        staleTime: 5_000,
        refetchInterval: 15_000,
      });
      return { view: q.data ?? null, loading: q.isLoading, error: q.error ?? null };
    }
