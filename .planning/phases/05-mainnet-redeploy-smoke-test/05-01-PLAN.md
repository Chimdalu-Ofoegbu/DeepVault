```bash
# scan_tbd_slots: awk walks the TOML, tracks current section header, applies
# section-aware exemption. Returns 0 if no offending TBD found; 1 + line list
# if any TBD remains in non-exempt section.
scan_tbd_slots() {
  local config_file="$1"
  local shipped="$2"  # "true" or "false"
  awk -v shipped="$shipped" '
    /^\[[a-z_]+\]/ { section = substr($0, 2, length($0)-2); next }
    /=.*"TBD"/ {
      # Exempt [deepvault] (filled by mainnet-deploy.sh later)
      if (section == "deepvault") next
      # Exempt [hosting] (Phase 6 scope per D-01)
      if (section == "hosting") next
      # Exempt [predict], [oracle_svi], [deepbook_margin] only when contingency=false
      if (shipped == "false" && (section == "predict" || section == "oracle_svi" || section == "deepbook_margin")) next
      print FILENAME ":" NR ": " section "." $0
      found = 1
    }
    END { exit found ? 1 : 0 }
  ' "$config_file"
}

echo "==> Step 13: config/mainnet.toml has no unfilled TBD slots (section-aware; exemptions: [hosting], [deepvault], and {[predict],[oracle_svi],[deepbook_margin]} when contingency.predict_mainnet_shipped=false)"
PREDICT_SHIPPED="$(awk '/^\[contingency\]/{flag=1; next} /^\[/{flag=0} flag && /^predict_mainnet_shipped\s*=/{ sub(/.*=\s*/,""); gsub(/[ \t]/,""); print; exit }' config/mainnet.toml)"
if [[ "${PREDICT_SHIPPED}" != "true" ]] && [[ "${PREDICT_SHIPPED}" != "false" ]]; then
  echo "::error::config/mainnet.toml [contingency].predict_mainnet_shipped must be true or false (got '${PREDICT_SHIPPED}')" >&2
  exit 1
fi

if ! scan_tbd_slots config/mainnet.toml "${PREDICT_SHIPPED}"; then
  echo "::error::config/mainnet.toml has unfilled TBD slots in non-exempt sections (see lines above)" >&2
  echo "         remediation: complete Phase 5 deploy steps OR (if Predict-mainnet-not-shipped) confirm [contingency].predict_mainnet_shipped=false" >&2
  exit 1
fi
```
