**Step 11 — WARNING 7 FIX: Patch config/mainnet.toml [deepvault] block via awk-based section-scoped helper (single approach; NO Python tomli / tomllib heredoc).**

Define the helper once at the top of the script (move this above Step 2 if preferred) and then invoke for each of the 5 [deepvault] keys:

```bash
# patch_toml_value: section-scoped, comment-preserving, atomic tempfile move.
# Replaces `${key} = "..."` in `[${section}]` with `${key} = "${value}"`.
# Single source of truth for all TOML patching in this script (WARNING 7 fix:
# no Python tomli/tomllib heredoc duplication).
patch_toml_value() {
  local section="$1" key="$2" value="$3"
  awk -v sec="\\[${section}\\]" -v k="^${key}\\s*=" -v v="\"${value}\"" '
    $0 ~ sec { flag=1; print; next }
    /^\[/ { flag=0 }
    flag && $0 ~ k { sub(/=.*$/, "= " v); print; next }
    { print }
  ' config/mainnet.toml > config/mainnet.toml.tmp && mv config/mainnet.toml.tmp config/mainnet.toml
}

# Patch all 5 [deepvault] fields:
patch_toml_value deepvault package_id "${PACKAGE_ID}"
patch_toml_value deepvault vault_shared_object_id "${VAULT_ID}"
patch_toml_value deepvault admin_cap_id "${ADMIN_CAP_ID}"
patch_toml_value deepvault treasury_cap_holder "${DEPLOYER_ADDR}"

  patch_toml_value oracle_svi top_level_shared_object_id "${ORACLE_SVI_ID}"
fi
```
