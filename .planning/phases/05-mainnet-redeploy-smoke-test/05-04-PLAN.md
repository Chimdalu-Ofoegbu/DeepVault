```typescript
async function snapshotVault(client: SuiClient, vaultId: string): Promise<SnapshotJson> {
    const obj = await client.getObject({ id: vaultId, options: { showContent: true } });
    const content = obj.data?.content;
    const fields = content && content.dataType === 'moveObject'
        ? ((content.fields as Record<string, unknown>) ?? {})
        : {};
    return {
        balance: String(fields.balance ?? '0'),
        total_assets: String(fields.total_assets ?? '0'),
        total_shares: String(fields.total_shares_supply ?? '0'),
    };
}

function navPerShareScaled1e9(snap: SnapshotJson): bigint {
    const ta = BigInt(snap.total_assets);
    const ts = BigInt(snap.total_shares);
    if (ts === 0n) return 0n;
    return (ta * 1_000_000_000n) / ts;
}

async function findDepositCoin(client: SuiClient, owner: string, type: string, minAmount: bigint): Promise<string> {
    const coins = await client.getCoins({ owner, coinType: type });
    const candidates = coins.data.filter((c) => BigInt(c.balance) >= minAmount);
    if (candidates.length === 0) {
        throw new Error(`No Coin<${type}> with balance >= ${minAmount} found for ${owner}`);
    }
    return candidates[0].coinObjectId;
}

# scripts/mainnet-smoke-test.sh
# Mainnet $50 USDsui round-trip orchestrator. Wraps scripts/mainnet-smoke-test.ts
# with deploy-status gate + structured output capture for Plan 05-06 docs append.
# Handles BOTH full-PTB path AND split-demo (D-05) path; the TS driver picks the
# path based on MAINNET-DEPLOY.json predict_package_id field (TBD = split-demo).
#
# DEPLOY-04 closure. Hard deadline 2026-06-12.
#
# Usage:
#   SUI_CONFIG_DIR=~/.sui/sui_config_mainnet bash scripts/mainnet-smoke-test.sh
#
# The TS driver takes ~60+ min wall-clock due to the 1h redeem cooldown.
# Exit codes:
#   0  All gates PASS + AdminCap re-verified (full-PTB) OR split-demo cycle complete
#   1  TS driver failed (see /tmp/smoke-test-$$.log)
#   2  Precondition failed (MAINNET-DEPLOY.json status != "deployed" or SUI_CONFIG_DIR wrong)

set -euo pipefail
REPO_ROOT="$(git rev-parse --show-toplevel)"
cd "$REPO_ROOT"

DEPLOY_JSON=".planning/phases/05-mainnet-redeploy-smoke-test/MAINNET-DEPLOY.json"

# Precondition 1: MAINNET-DEPLOY.json exists and is status=deployed
if [[ ! -f "${DEPLOY_JSON}" ]]; then
  echo "::error::${DEPLOY_JSON} not found. Run scripts/mainnet-deploy.sh first." >&2
  exit 2
fi
DEPLOY_STATUS="$(jq -r '.status // "unknown"' "${DEPLOY_JSON}")"
if [[ "${DEPLOY_STATUS}" != "deployed" ]]; then
  echo "::error::MAINNET-DEPLOY.json status='${DEPLOY_STATUS}'; expected 'deployed'. Run scripts/mainnet-deploy.sh." >&2
  exit 2
fi

# Precondition 2: SUI_CONFIG_DIR scoping (Phase 0 D-06)
if [[ -z "${SUI_CONFIG_DIR:-}" ]] || [[ "${SUI_CONFIG_DIR}" != *"sui_config_mainnet"* ]]; then
  echo "::error::SUI_CONFIG_DIR must end in 'sui_config_mainnet'; remediation: SUI_CONFIG_DIR=~/.sui/sui_config_mainnet bash scripts/mainnet-smoke-test.sh" >&2
  exit 2
fi

# Precondition 3: Active env is mainnet
ENV_NAME="$(sui client active-env)"
if [[ "${ENV_NAME}" != "mainnet" ]]; then
  echo "::error::Active sui env is '${ENV_NAME}', expected 'mainnet'" >&2
  exit 2
fi

# Detect path mode (for operator banner)
PREDICT_PKG="$(jq -r '.predict_package_id // "TBD"' "${DEPLOY_JSON}")"
if [[ "${PREDICT_PKG}" == "TBD" ]]; then
  echo "==> Path mode: SPLIT-DEMO (D-05) — predict_package_id=TBD; running supply/redeem-only via AdminCap admin_tune_strategy(0)"
else
  echo "==> Path mode: FULL-PTB — predict_package_id=${PREDICT_PKG}; running full supply/hedge/redeem cycle"
fi
echo "==> All preconditions green. Starting smoke test (~60+ min wall-clock)..."
echo "==> Pre-deposit time: $(date -u +%Y-%m-%dT%H:%M:%SZ)"

SMOKE_LOG="/tmp/smoke-test-$$.log"
trap "echo '==> Log: ${SMOKE_LOG}'" EXIT

cd "${REPO_ROOT}/dashboard"
set +e
npx tsx ../scripts/mainnet-smoke-test.ts 2>&1 | tee "${SMOKE_LOG}"
SMOKE_EXIT=${PIPESTATUS[0]}
set -e

cd "${REPO_ROOT}"

if [[ ${SMOKE_EXIT} -ne 0 ]]; then
  echo "::error::Smoke test TS driver exited ${SMOKE_EXIT}. See ${SMOKE_LOG} for details." >&2
  echo "::error::Recovery: see docs/MAINNET-FUNDING.md Recovery section. If redeem_request succeeded but redeem_fulfill failed, run scripts/redeem-cancel.sh to recover escrowed shares (manual). If split-demo allocation_bps was tuned to 0 but not restored, restore via AdminCap admin_tune_strategy with the prior value before any further activity." >&2
  exit ${SMOKE_EXIT}
fi

# Parse structured receipts for Plan 05-06 docs append (both paths)
echo ""
echo "================================================================"
echo "SMOKE TEST GREEN — Receipts (input to docs/MAINNET-FUNDING.md update):"
echo "================================================================"
grep -E '^(supply_tx_digest|redeem_request_tx_digest|redeem_fulfill_tx_digest|hedge_cost_basis_quote|received_quote|pre_deposit_nav_1e9|post_redeem_nav_1e9|gate_per_depositor_ratio|gate_vault_nav_delta_bps|admin_cap_owner_verified|hedge_market_key|hedge_strike|hedge_expiry|split_demo_.*|admin_tune_.*)=' "${SMOKE_LOG}" \
  | tee "${REPO_ROOT}/.planning/phases/05-mainnet-redeploy-smoke-test/smoke-receipts.txt"

echo ""
echo "Receipts file: .planning/phases/05-mainnet-redeploy-smoke-test/smoke-receipts.txt"
if [[ "${PREDICT_PKG}" == "TBD" ]]; then
  echo "Split-demo artifact: .planning/phases/05-mainnet-redeploy-smoke-test/SMOKE-TEST-SPLIT-DEMO.json"
fi
echo "Next step: Plan 05-06 docs append (operator manually or via helper)"
exit 0
```
